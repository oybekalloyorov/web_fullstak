Node.js va npm

Node.js nima, modullar tizimi (CommonJS va ESM), npm va package.json, fs, path, events, http modullari, muhit o'zgaruvchilari va CLI dasturlar.

O'rta · ~100 daqiqa

Shu paytgacha JavaScript faqat brauzerda ishladi. Node.js — JavaScript'ni brauzerdan tashqarida, ya'ni kompyuter yoki serverda ishga tushiruvchi muhit (runtime). U tufayli siz bilgan til bilan web-serverlar, API'lar, CLI dasturlar va avtomatlashtirish skriptlarini yoza olasiz. Bu dars — backend olamiga kirish eshigi.

#Node.js nima va qanday ishlaydi?

  • 2009-yilda Ryan Dahl yaratgan. Ichida Google Chrome'ning V8 JavaScript dvigateli ishlaydi.
  • libuv kutubxonasi orqali fayl tizimi, tarmoq, taymerlar bilan bloklanmaydigan (non-blocking) tarzda ishlaydi.
  • Brauzerdagi kabi event loop modeliga asoslangan (6-dars): bitta oqim minglab ulanishlarga xizmat qiladi, chunki kutish paytida (bazadan javob, fayl o'qish) boshqa so'rovlarni qayta ishlaydi.

#http moduli — freymvorksiz server

Express'ga o'tishdan oldin, uning ostida nima borligini ko'raylik:

server.js
import http from 'node:http';

const server = http.createServer(async (req, res) => {
  const url = new URL(req.url, `http://${req.headers.host}`);

  // So'rov tanasini o'qish — stream bo'laklarini yig'amiz
  let body = '';
  for await (const chunk of req) body += chunk;

  console.log(req.method, url.pathname, url.searchParams.toString(), body);

  res.setHeader('Content-Type', 'application/json; charset=utf-8');

  if (req.method === 'GET' && url.pathname === '/api/salom') {
    const ism = url.searchParams.get('ism') ?? 'Mehmon';
    res.end(JSON.stringify({ xabar: `Salom, ${ism}!` }));
  } else if (req.method === 'POST' && url.pathname === '/api/echo') {
    try {
      const data = JSON.parse(body);
      res.statusCode = 201;
      res.end(JSON.stringify({ olindi: data }));
    } catch {
      res.statusCode = 400;
      res.end(JSON.stringify({ xato: 'JSON noto\'g\'ri' }));
    }
  } else {
    res.statusCode = 404;
    res.end(JSON.stringify({ xato: 'Topilmadi' }));
  }
});

server.listen(3000, () => console.log('http://localhost:3000'));

Ko'rib turganingizdek: marshrutlash (if/else), tanani o'qish, JSON'ni parse qilish, xatolar — hammasi qo'lda. Marshrutlar 50 taga yetsa, bu kod boshqarib bo'lmaydigan bo'ladi. Aynan shu muammoni Express hal qiladi (9-dars).

"Downloads" papkasidagi yuzlab fayllarni turiga qarab papkalarga ajratadigan foydali skript.

organize.js
#!/usr/bin/env node
import fs from 'node:fs/promises';
import path from 'node:path';

const TOIFALAR = {
  Rasmlar: ['.jpg', '.jpeg', '.png', '.gif', '.webp', '.svg'],
  Hujjatlar: ['.pdf', '.doc', '.docx', '.xls', '.xlsx', '.ppt', '.pptx', '.txt'],
  Videolar: ['.mp4', '.mkv', '.avi', '.mov'],
  Musiqa: ['.mp3', '.wav', '.flac'],
  Arxivlar: ['.zip', '.rar', '.7z', '.tar', '.gz'],
  Dasturlar: ['.exe', '.msi', '.dmg', '.deb'],
};

// Kengaytma → toifa (tez qidirish uchun teskari jadval)
const kengaytmaToifa = Object.fromEntries(
  Object.entries(TOIFALAR).flatMap(([toifa, exts]) => exts.map(e => [e, toifa]))
);

const [, , papka = '.', ...flags] = process.argv;
const sinov = flags.includes('--dry-run');   // faqat ko'rsatish, ko'chirmaslik

async function tartibla(dir) {
  const elementlar = await fs.readdir(dir, { withFileTypes: true });
  const hisobot = {};

  for (const el of elementlar) {
    if (!el.isFile() || el.name.startsWith('.')) continue;

    const toifa = kengaytmaToifa[path.extname(el.name).toLowerCase()] ?? 'Boshqalar';
    const manzilPapka = path.join(dir, toifa);
    let manzil = path.join(manzilPapka, el.name);

    // Bir xil nomli fayl bo'lsa: rasm.png → rasm (1).png
    for (let i = 1; await mavjudmi(manzil); i++) {
      const { name, ext } = path.parse(el.name);
      manzil = path.join(manzilPapka, `${name} (${i})${ext}`);
    }

    if (!sinov) {
      await fs.mkdir(manzilPapka, { recursive: true });
      await fs.rename(path.join(dir, el.name), manzil);
    }
    hisobot[toifa] = (hisobot[toifa] ?? 0) + 1;
    console.log(`${sinov ? '[sinov] ' : ''}${el.name} → ${toifa}/`);
  }
  return hisobot;
}

async function mavjudmi(p) {
  try { await fs.access(p); return true; } catch { return false; }
}

try {
  const natija = await tartibla(path.resolve(papka));
  console.log('\n📊 Natija:');
  console.table(natija);
} catch (err) {
  console.error(`❌ Xato: ${err.code === 'ENOENT' ? 'papka topilmadi' : err.message}`);
  process.exit(1);
}

Statik fayl serveri (http + fs + path)

Shaxsiy saytingizni (2–3-darslar) brauzerda http://localhost:8080 orqali ochadigan server. Unda real serverlar duch keladigan muammolar — MIME turlari, 404 va xavfsizlik — hal qilingan.

static-server.js
import http from 'node:http';
import fs from 'node:fs/promises';
import { createReadStream } from 'node:fs';
import path from 'node:path';

const ROOT = path.resolve(process.argv[2] ?? 'public');
const PORT = Number(process.env.PORT) || 8080;

const MIME = {
  '.html': 'text/html; charset=utf-8',
  '.css': 'text/css; charset=utf-8',
  '.js': 'text/javascript; charset=utf-8',
  '.json': 'application/json',
  '.png': 'image/png',
  '.jpg': 'image/jpeg',
  '.svg': 'image/svg+xml',
  '.ico': 'image/x-icon',
  '.woff2': 'font/woff2',
};

const server = http.createServer(async (req, res) => {
  const started = Date.now();
  res.on('finish', () =>
    console.log(`${req.method} ${req.url} → ${res.statusCode} (${Date.now() - started}ms)`)
  );
  try {
    const { pathname } = new URL(req.url, 'http://localhost');
    let filePath = path.join(ROOT, decodeURIComponent(pathname));

    // 🔒 Path traversal hujumidan himoya: /../../etc/passwd
    if (filePath !== ROOT && !filePath.startsWith(ROOT + path.sep)) {
      res.writeHead(403).end('Forbidden');
      return;
    }

    let stat = await fs.stat(filePath).catch(() => null);
    if (stat?.isDirectory()) {
      filePath = path.join(filePath, 'index.html');
      stat = await fs.stat(filePath).catch(() => null);
    }

    if (!stat) {
      res.writeHead(404, { 'Content-Type': 'text/html; charset=utf-8' });
      res.end('<h1>404</h1><p>Sahifa topilmadi</p>');
      return;
    }

    res.writeHead(200, {
      'Content-Type': MIME[path.extname(filePath)] ?? 'application/octet-stream',
      'Content-Length': stat.size,
      'Cache-Control': 'public, max-age=3600',
    });
    createReadStream(filePath).pipe(res);   // stream — katta fayllar ham xotirani to'ldirmaydi
  } catch (err) {
    console.error(err);
    res.writeHead(500).end('Server xatosi');
  }
});

server.listen(PORT, () => console.log(`📂 ${ROOT} → http://localhost:${PORT}`));

JSON faylga asoslangan "mini ma'lumotlar bazasi" moduli

Haqiqiy bazaga (10-dars) o'tishdan oldin, ma'lumotni JSON faylda saqlovchi qayta ishlatiladigan modul yozamiz. U 9-darsdagi Express API'da ishlatiladi.

src/db/jsonStore.js
import fs from 'node:fs/promises';
import path from 'node:path';
import { randomUUID } from 'node:crypto';

export class JsonStore {
  #file;
  #queue = Promise.resolve();     // yozuvlarni navbatga qo'yish uchun

  constructor(file) {
    this.#file = path.resolve(file);
  }

  async #read() {
    try {
      return JSON.parse(await fs.readFile(this.#file, 'utf8'));
    } catch (err) {
      if (err.code === 'ENOENT') return [];   // fayl hali yo'q
      throw err;
    }
  }

  // Bir vaqtdagi ikki yozuv bir-birini o'chirib yubormasligi uchun — ketma-ket bajaramiz
  #write(mutator) {
    this.#queue = this.#queue.then(async () => {
      const items = await this.#read();
      const result = mutator(items);
      const tmp = `${this.#file}.tmp`;
      await fs.mkdir(path.dirname(this.#file), { recursive: true });
      await fs.writeFile(tmp, JSON.stringify(items, null, 2));
      await fs.rename(tmp, this.#file);        // atomar almashtirish — fayl hech qachon yarim yozilmaydi
      return result;
    });
    return this.#queue;
  }

  async all({ filter = () => true, sort } = {}) {
    const items = (await this.#read()).filter(filter);
    return sort ? items.sort(sort) : items;
  }

  async get(id) {
    return (await this.#read()).find(i => i.id === id) ?? null;
  }

  create(data) {
    return this.#write(items => {
      const item = { id: randomUUID(), ...data, createdAt: new Date().toISOString() };
      items.push(item);
      return item;
    });
  }

  update(id, patch) {
    return this.#write(items => {
      const i = items.findIndex(x => x.id === id);
      if (i === -1) return null;
      items[i] = { ...items[i], ...patch, id, updatedAt: new Date().toISOString() };
      return items[i];
    });
  }

  remove(id) {
    return this.#write(items => {
      const i = items.findIndex(x => x.id === id);
      if (i === -1) return false;
      items.splice(i, 1);
      return true;
    });
  }
}
try.js
