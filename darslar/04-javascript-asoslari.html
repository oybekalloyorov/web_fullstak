Closure (yopilish)

Closure — funksiya o'zi yaratilgan muhitdagi o'zgaruvchilarni, hatto o'sha muhit tugaganidan keyin ham eslab qolishi.

JavaScript
function hisoblagichYarat() {
  let son = 0;                 // tashqaridan to'g'ridan-to'g'ri o'zgartirib bo'lmaydi (xususiy)
  return {
    oshir: () => ++son,
    kamaytir: () => --son,
    qiymat: () => son,
  };
}

const h1 = hisoblagichYarat();

JavaScript
class Hisob {
  #balans = 0;                      // # — haqiqiy xususiy (private) maydon
  static bankNomi = 'Ishonch Bank'; // sinfga tegishli, obyektga emas

  constructor(egasi, boshlangich = 0) {
    this.egasi = egasi;
    this.#balans = boshlangich;
    this.tarix = [];
  }

  get balans() {                    // getter — xususiyat kabi o'qiladi: hisob.balans
    return this.#balans;
  }

  toldirish(summa) {
    if (summa <= 0) throw new Error('Summa musbat bo\'lishi kerak');
    this.#balans += summa;
    this.tarix.push({ tur: 'kirim', summa, sana: new Date() });
    return this;                    // zanjir uchun
  }

  yechish(summa) {
    if (summa > this.#balans) throw new Error('Mablag\' yetarli emas');
    this.#balans -= summa;
    this.tarix.push({ tur: 'chiqim', summa, sana: new Date() });
    return this;
  }
}

// Meros (inheritance)
class JamgarmaHisob extends Hisob {
  constructor(egasi, foiz) {
    super(egasi);                   // ota konstruktorini chaqirish majburiy
    this.foiz = foiz;
  }
  foizQosh() {
    return this.toldirish(this.balans * this.foiz / 100);
  }
}

const h = new JamgarmaHisob('Laylo', 12);

function promokodQoll(summa, kod) {

  const qatorlar = savat.map(m => ({ ...m, jami: qatorSummasi(m) }));
  const oraliq = qatorlar.reduce((s, q) => s + q.jami, 0);
  const tejaldi = savat.reduce((s, m) => s + m.narx * m.soni, 0) - oraliq;
  const { summa: promodan, xabar } = promokodQoll(oraliq, promokod);
  const yetkazish = promodan >= BEPUL_YETKAZISH_CHEGARASI ? 0 : YETKAZISH_NARXI;
  const jami = promodan + yetkazish;
  const mahsulotSoni = savat.reduce((s, m) => s + m.soni, 0);

  return { qatorlar, oraliq, tejaldi, promokodXabari: xabar, yetkazish, jami, mahsulotSoni };
}

const f = (n) => `${n.toLocaleString('uz-UZ')} so'm`;
const natija = hisobla(savat, 'bahor10');

console.table(natija.qatorlar.map(({ nomi, soni, jami }) => ({ nomi, soni, jami: f(jami) })));
console.log(`Mahsulotlar: ${natija.mahsulotSoni} ta`);
console.log(`Oraliq summa: ${f(natija.oraliq)} (chegirmalar: -${f(natija.tejaldi)})`);
console.log(natija.promokodXabari);
console.log(`Yetkazish: ${natija.yetkazish ? f(natija.yetkazish) : 'BEPUL'}`);
console.log(`To'lovga: ${f(natija.jami)}`);

Nimani o'rgandik: map/reduce bilan hisob-kitob, destructuring, standart parametrlar, obyektdan qidiruv-jadval (lookup table) sifatida foydalanish, sof funksiyalar (bir xil kirish → bir xil chiqish), raqamli ajratgich (450_000). Bu kod 13-darsda React interfeysiga ulanadi.

Closure bilan amaliy vositalar: memoize, once va debounce

Professional kodda tez-tez uchraydigan uchta yordamchi funksiya — barchasi closure asosida.

helpers.js
// 1. memoize — og'ir hisob natijasini keshlash
function memoize(fn) {
  const kesh = new Map();                 // closure ichida saqlanadi
  return function (...args) {
    const kalit = JSON.stringify(args);
    if (kesh.has(kalit)) return kesh.get(kalit);
    const natija = fn(...args);
    kesh.set(kalit, natija);
    return natija;
  };
}

const fib = memoize(n => (n < 2 ? n : fib(n - 1) + fib(n - 2)));
console.time('fib');
console.log(fib(90));   // 2880067194370816000 — bir zumda (keshsiz yillab hisoblanardi)
console.timeEnd('fib');

// 2. once — funksiyani faqat bir marta bajarish (masalan "To'lash" tugmasi)
function once(fn) {
  let bajarildi = false, natija;
  return (...args) => {
    if (!bajarildi) {
      bajarildi = true;
      natija = fn(...args);
    }
    return natija;
  };
}
const tolash = once((summa) => { console.log(`${summa} so'm to'landi`); return 'OK'; });
tolash(100000);  // "100000 so'm to'landi"
tolash(100000);  // hech narsa — ikkinchi marta bajarilmaydi

// 3. debounce — ketma-ket chaqiruvlarda faqat oxirgisini bajarish
//    (qidiruv maydoniga yozilganda har harf uchun emas, yozib bo'lgach so'rov yuborish)
function debounce(fn, kutish = 300) {
  let timer;
  return (...args) => {
    clearTimeout(timer);
    timer = setTimeout(() => fn(...args), kutish);
  };
}
const qidir = debounce((soz) => console.log('Serverga so\'rov:', soz), 500);
qidir('j'); qidir('ja'); qidir('jav'); qidir('java');
// 500ms dan keyin faqat bitta: "Serverga so'rov: java"

Uchala funksiya ham ichki holatni (kesh, bajarildi, timer) closure orqali saqlaydi. debounce ni 6-darsda real qidiruv bilan ishlatamiz.

Kutubxona boshqaruv tizimi (sinflar, xatolar, Map)

library.js
class KitobMavjudEmasError extends Error {
  constructor(isbn) {
    super(`ISBN ${isbn} bo'yicha bo'sh nusxa yo'q`);
    this.name = 'KitobMavjudEmasError';
  }
}

class Kitob {
  constructor({ isbn, nomi, muallif, nusxalar = 1 }) {
    Object.assign(this, { isbn, nomi, muallif });
    this.jami = nusxalar;
    this.bosh = nusxalar;
  }
  get mavjudmi() { return this.bosh > 0; }
  toString() { return `"${this.nomi}" — ${this.muallif} (${this.bosh}/${this.jami})`; }
}

class Kutubxona {
  #kitoblar = new Map();      // isbn → Kitob
  #ijaralar = [];             // { isbn, oquvchi, berilgan, muddat }
  static IJARA_KUNI = 14;
  static JARIMA_KUNIGA = 2000;

  qoshish(malumot) {
    const mavjud = this.#kitoblar.get(malumot.isbn);
    if (mavjud) {
      mavjud.jami += malumot.nusxalar ?? 1;
      mavjud.bosh += malumot.nusxalar ?? 1;
    } else {
      this.#kitoblar.set(malumot.isbn, new Kitob(malumot));
    }
    return this;
  }

  qidirish(soz) {
    const q = soz.toLowerCase();
    return [...this.#kitoblar.values()].filter(
      k => k.nomi.toLowerCase().includes(q) || k.muallif.toLowerCase().includes(q)
    );
  }

  berish(isbn, oquvchi, sana = new Date()) {
    const kitob = this.#kitoblar.get(isbn);
    if (!kitob) throw new Error(`Bunday kitob yo'q: ${isbn}`);
    if (!kitob.mavjudmi) throw new KitobMavjudEmasError(isbn);
    kitob.bosh--;
    const muddat = new Date(sana);
    muddat.setDate(muddat.getDate() + Kutubxona.IJARA_KUNI);
    this.#ijaralar.push({ isbn, oquvchi, berilgan: sana, muddat });
    return muddat;
  }

  qaytarish(isbn, oquvchi, sana = new Date()) {
    const i = this.#ijaralar.findIndex(r => r.isbn === isbn && r.oquvchi === oquvchi);
    if (i === -1) throw new Error('Bunday ijara topilmadi');
    const [ijara] = this.#ijaralar.splice(i, 1);
    this.#kitoblar.get(isbn).bosh++;
    const kechikish = Math.ceil((sana - ijara.muddat) / 86_400_000);
    return kechikish > 0 ? kechikish * Kutubxona.JARIMA_KUNIGA : 0;
  }

  get hisobot() {
    const kitoblar = [...this.#kitoblar.values()];
    return {
      nomlar: kitoblar.length,
      jamiNusxa: kitoblar.reduce((s, k) => s + k.jami, 0),
      ijarada: this.#ijaralar.length,
      tugaganlar: kitoblar.filter(k => !k.mavjudmi).map(String),
    };
  }
}

}
