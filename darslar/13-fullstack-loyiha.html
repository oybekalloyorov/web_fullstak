Yakuniy loyiha: full-stack "Eslatmalar" ilovasi

Barcha bilimlarni birlashtiramiz: PostgreSQL + Express API + JWT autentifikatsiya + React frontend. Loyihalash, backend, frontend va ularni bog'lash bosqichma-bosqich.

Yuqori · ~4–6 soat amaliyot
📄 PDF yuklab olish

Endi barcha bo'laklarni bitta haqiqiy ilovaga yig'amiz. "Eslatmalar" (Notes) — foydalanuvchilar ro'yxatdan o'tadi, shaxsiy eslatmalar yaratadi, ularni teglar bilan belgilaydi, qidiradi, mahkamlaydi (pin) va arxivlaydi. Loyiha kichik ko'rinadi, lekin unda real ishdagi barcha muammolar bor: sxema dizayni, API, autentifikatsiya, egalik, validatsiya, frontend holati, xatolar, CORS va proksi.

Qanday o'qish kerak Bu dars — qo'llanma emas, amaliy ish. Har bir faylni o'zingiz yarating va har bosqich oxirida ilovani ishga tushirib tekshiring. Tushunmagan joyingiz bo'lsa — tegishli darsga qayting (havolalar matnda).

#1-bosqich: Loyihalash

Funksional talablar

  • Ro'yxatdan o'tish, kirish, chiqish; sessiya sahifa yangilanganda saqlanadi.
  • Eslatma: sarlavha, matn, rang, teglar, "mahkamlangan" va "arxivlangan" belgilari.
  • Eslatmalarni yaratish, tahrirlash, o'chirish (CRUD); faqat o'z eslatmalarini ko'rish.
  • Matn bo'yicha qidiruv, teg bo'yicha filtr, arxiv bo'limi; mahkamlanganlar tepada.

#2-bosqich: Ma'lumotlar bazasi

server/db/schema.sql
CREATE EXTENSION IF NOT EXISTS pgcrypto;   -- gen_random_uuid() uchun (PG 13+ da o'rnatilgan)

CREATE TABLE IF NOT EXISTS users (
  id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name          TEXT NOT NULL CHECK (length(name) BETWEEN 2 AND 60),
  email         TEXT NOT NULL,
  password_hash TEXT NOT NULL,
  created_at    TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE UNIQUE INDEX IF NOT EXISTS users_email_uq ON users (lower(email));

CREATE TABLE IF NOT EXISTS notes (
  id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id     UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  title       TEXT NOT NULL DEFAULT '' CHECK (length(title) <= 200),
  body        TEXT NOT NULL DEFAULT '' CHECK (length(body) <= 20000),
  color       TEXT NOT NULL DEFAULT 'default',
  tags        TEXT[] NOT NULL DEFAULT '{}',
  pinned      BOOLEAN NOT NULL DEFAULT false,
  archived    BOOLEAN NOT NULL DEFAULT false,
  created_at  TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at  TIMESTAMPTZ NOT NULL DEFAULT now()
);
-- Eng ko'p so'rov: "foydalanuvchining arxivlanmagan eslatmalari, mahkamlanganlar va yangilari tepada"
CREATE INDEX IF NOT EXISTS notes_user_list_idx ON notes (user_id, archived, pinned DESC, updated_at DESC);
CREATE INDEX IF NOT EXISTS notes_tags_idx ON notes USING GIN (tags);   -- massiv ichida qidiruv uchun

Teglar uchun alohida N:M jadval o'rniga PostgreSQL massivi (TEXT[]) tanlandi: teglar faqat bitta foydalanuvchiga tegishli, alohida xususiyatlari yo'q, va GIN indeksi tags @> ARRAY['ish'] so'rovini tezlashtiradi. Bu — loyihaning ehtiyojiga qarab qilingan ongli tanlov (10-dars).

export class HttpError extends Error {
  constructor(status, message, details) {
    super(message);
    this.status = status;
    this.details = details;
  }
}

export function notFound(req, res, next) {
  next(new HttpError(404, `Topilmadi: ${req.method} ${req.originalUrl}`));
}

export function errorHandler(err, req, res, _next) {
  if (err.type === 'entity.parse.failed') err = new HttpError(400, 'JSON noto\'g\'ri');
  if (err.code === '22P02') err = new HttpError(404, 'Topilmadi');          // noto'g'ri UUID
  const status = err.status ?? 500;
  if (status >= 500) console.error(err);
  res.status(status).json({
    error: {
      message: status >= 500 && process.env.NODE_ENV === 'production' ? 'Server xatosi' : err.message,
      ...(err.details && { details: err.details }),
    },
  });
}
server/src/middleware/validate.js
import { HttpError } from './errors.js';

export const validate = (schema, source = 'body') => (req, res, next) => {
  const r = schema.safeParse(req[source]);
  if (!r.success) {
    return next(new HttpError(400, 'Validatsiya xatosi',
      r.error.issues.map(i => ({ field: i.path.join('.'), message: i.message }))));
  }
  req.valid = r.data;
  next();
};
server/src/middleware/auth.js
import jwt from 'jsonwebtoken';
import { config } from '../config.js';
import { HttpError } from './errors.js';

export const COOKIE = 'notes_token';

export const cookieOptions = {
  httpOnly: true,
  secure: config.isProd,
  sameSite: 'lax',
  maxAge: 7 * 24 * 60 * 60 * 1000,      // 7 kun (soddalik uchun refresh token'siz)
  path: '/',
};

export const signToken = (user) =>
  jwt.sign({ sub: user.id, name: user.name }, config.jwtSecret, { expiresIn: '7d', algorithm: 'HS256' });

export function requireAuth(req, res, next) {
  const token = req.cookies[COOKIE];
  if (!token) return next(new HttpError(401, 'Tizimga kiring'));
  try {
    const payload = jwt.verify(token, config.jwtSecret, { algorithms: ['HS256'] });
    req.userId = payload.sub;
    next();
  } catch {
    next(new HttpError(401, 'Sessiya muddati tugagan'));
  }
}

import { Router } from 'express';
import bcrypt from 'bcrypt';
import { z } from 'zod';
import { query } from '../db.js';
import { validate } from '../middleware/validate.js';
import { HttpError } from '../middleware/errors.js';
import { COOKIE, cookieOptions, signToken, requireAuth } from '../middleware/auth.js';

const router = Router();
const DUMMY = await bcrypt.hash('dummy', 12);

const registerSchema = z.object({
  name: z.string().trim().min(2, 'Ism kamida 2 harf').max(60),
  email: z.string().trim().toLowerCase().email('Email noto\'g\'ri'),
  password: z.string().min(8, 'Parol kamida 8 belgi').max(72),
});
const loginSchema = z.object({
  email: z.string().trim().toLowerCase().email(),
  password: z.string().min(1),
});

const login = (res, user, status = 200) => {
  res.cookie(COOKIE, signToken(user), cookieOptions);
  res.status(status).json({ user: { id: user.id, name: user.name, email: user.email } });
};

router.post('/register', validate(registerSchema), async (req, res) => {
  const { name, email, password } = req.valid;
  const hash = await bcrypt.hash(password, 12);
  try {
    const { rows: [user] } = await query(
      'INSERT INTO users (name, email, password_hash) VALUES ($1, $2, $3) RETURNING id, name, email',
      [name, email, hash]
    );
    login(res, user, 201);
  } catch (err) {
    if (err.code === '23505') throw new HttpError(409, 'Bu email band');
    throw err;
  }
});

router.post('/login', validate(loginSchema), async (req, res) => {
  const { email, password } = req.valid;
  const { rows: [user] } = await query('SELECT * FROM users WHERE lower(email) = $1', [email]);
  const ok = await bcrypt.compare(password, user?.password_hash ?? DUMMY);
  if (!user || !ok) throw new HttpError(401, 'Email yoki parol noto\'g\'ri');
  login(res, user);
});

router.post('/logout', (req, res) => {
  const { maxAge, ...opts } = cookieOptions;
  res.clearCookie(COOKIE, opts).status(204).end();
});

router.get('/me', requireAuth, async (req, res) => {
  const { rows: [user] } = await query('SELECT id, name, email FROM users WHERE id = $1', [req.userId]);
  if (!user) throw new HttpError(401, 'Foydalanuvchi topilmadi');
  res.json({ user });
});

export default router;
server/src/routes/notes.routes.js
import { Router } from 'express';
import { z } from 'zod';
import { query } from '../db.js';
import { validate } from '../middleware/validate.js';
import { HttpError } from '../middleware/errors.js';
import { requireAuth } from '../middleware/auth.js';

const router = Router();
router.use(requireAuth);                         // ⬅ bu router'dagi BARCHA marshrutlar himoyalangan

const COLORS = ['default', 'yellow', 'green', 'blue', 'pink', 'purple'];
const tag = z.string().trim().toLowerCase().min(1).max(30).regex(/^[\p{L}\p{N}_-]+$/u, 'Teg: harf, raqam, _ yoki -');

const noteSchema = z.object({
  title: z.string().trim().max(200).default(''),
  body: z.string().max(20000).default(''),
  color: z.enum(COLORS).default('default'),
  tags: z.array(tag).max(10).default([]).transform(t => [...new Set(t)]),   // takrorlarni olib tashlash
  pinned: z.boolean().default(false),
  archived: z.boolean().default(false),
}).refine(n => n.title || n.body, { message: 'Sarlavha yoki matn bo\'lishi kerak' });

const patchSchema = z.object({
  title: z.string().trim().max(200),
  body: z.string().max(20000),
  color: z.enum(COLORS),
  tags: z.array(tag).max(10).transform(t => [...new Set(t)]),
  pinned: z.boolean(),
  archived: z.boolean(),
}).partial().strict();          // strict — noma'lum maydon (masalan user_id) yuborilsa xato

const listSchema = z.object({
  q: z.string().trim().max(100).optional(),
  tag: tag.optional(),
  archived: z.enum(['true', 'false']).default('false').transform(v => v === 'true'),
});

const COLS = `id, title, body, color, tags, pinned, archived,
              created_at AS "createdAt", updated_at AS "updatedAt"`;

// GET /api/notes?q=&tag=&archived=
router.get('/', validate(listSchema, 'query'), async (req, res) => {
  const { q, tag, archived } = req.valid;
  const params = [req.userId, archived];
  let sql = `SELECT ${COLS} FROM notes WHERE user_id = $1 AND archived = $2`;

  if (q) {
    params.push(`%${q}%`);
    sql += ` AND (title ILIKE $${params.length} OR body ILIKE $${params.length})`;
  }
  if (tag) {
    params.push([tag]);
    sql += ` AND tags @> $${params.length}`;
  }
  sql += ' ORDER BY pinned DESC, updated_at DESC LIMIT 500';

  const { rows } = await query(sql, params);
  res.json({ data: rows });
});

router.post('/', validate(noteSchema), async (req, res) => {
  const n = req.valid;
  const { rows: [note] } = await query(
    `INSERT INTO notes (user_id, title, body, color, tags, pinned, archived)
     VALUES ($1, $2, $3, $4, $5, $6, $7) RETURNING ${COLS}`,
    [req.userId, n.title, n.body, n.color, n.tags, n.pinned, n.archived]
  );
  res.status(201).json(note);
});

router.patch('/:id', validate(patchSchema), async (req, res) => {
  const entries = Object.entries(req.valid);
  if (!entries.length) throw new HttpError(400, 'O\'zgartirish uchun maydon yo\'q');

  // Ustun nomlari — faqat sxema ruxsat bergan kalitlar (strict), qiymatlar — parametrlar
  const set = entries.map(([k], i) => `${k} = $${i + 3}`).join(', ');
  const { rows: [note] } = await query(
    `UPDATE notes SET ${set}, updated_at = now()
     WHERE id = $1 AND user_id = $2                -- 🔒 egalik tekshiruvi (IDOR'ga qarshi)
     RETURNING ${COLS}`,
    [req.params.id, req.userId, ...entries.map(([, v]) => v)]
  );
  if (!note) throw new HttpError(404, 'Eslatma topilmadi');
  res.json(note);
});

router.delete('/:id', async (req, res) => {
  const { rowCount } = await query('DELETE FROM notes WHERE id = $1 AND user_id = $2', [req.params.id, req.userId]);
  if (!rowCount) throw new HttpError(404, 'Eslatma topilmadi');
  res.status(204).end();
});

export default router;

// Teglar statistikasi alohida router'da
export const tagsRouter = Router().get('/', requireAuth, async (req, res) => {
  const { rows } = await query(
    `SELECT t AS name, COUNT(*)::int AS count
     FROM notes, unnest(tags) AS t
     WHERE user_id = $1 AND NOT archived
     GROUP BY t ORDER BY count DESC, t`,
    [req.userId]
  );
  res.json(rows);
});

export class ApiError extends Error {
  constructor(status, message, details) {
    super(message);
    this.status = status;
    this.details = details;
  }
}

export async function api(path, { method = 'GET', body, signal } = {}) {
  const res = await fetch('/api' + path, {
    method,
    signal,
    credentials: 'same-origin',
    headers: body ? { 'Content-Type': 'application/json' } : undefined,
    body: body ? JSON.stringify(body) : undefined,
  });
  if (res.status === 204) return null;
  const data = await res.json().catch(() => ({}));
  if (!res.ok) throw new ApiError(res.status, data.error?.message ?? 'Tarmoq xatosi', data.error?.details);
  return data;
}

export const authApi = {
  me: () => api('/auth/me'),
  login: (body) => api('/auth/login', { method: 'POST', body }),
  register: (body) => api('/auth/register', { method: 'POST', body }),
  logout: () => api('/auth/logout', { method: 'POST' }),
};

client/src/context/AuthContext.jsx
import { createContext, useContext, useEffect, useState } from 'react';
import { authApi } from '../api';

const AuthContext = createContext(null);

export function AuthProvider({ children }) {
  const [user, setUser] = useState(null);
  const [checking, setChecking] = useState(true);   // sahifa ochilganda sessiyani tekshirish

  useEffect(() => {
    authApi.me()
      .then(d => setUser(d.user))
      .catch(() => setUser(null))
      .finally(() => setChecking(false));
  }, []);

  const value = {
    user,
    checking,
    login: async (creds) => setUser((await authApi.login(creds)).user),
    register: async (data) => setUser((await authApi.register(data)).user),
    logout: async () => { await authApi.logout(); setUser(null); },
  };

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

export const useAuth = () => useContext(AuthContext);

import { BrowserRouter, Routes, Route, Navigate } from 'react-router';
import { AuthProvider, useAuth } from './context/AuthContext';
import LoginPage from './pages/LoginPage';
import NotesPage from './pages/NotesPage';

function Protected({ children }) {
  const { user, checking } = useAuth();
  if (checking) return <div className="center">⏳</div>;
  return user ? children : <Navigate to="/login" replace />;
}

function GuestOnly({ children }) {
  const { user, checking } = useAuth();
  if (checking) return null;
  return user ? <Navigate to="/" replace /> : children;
}

export default function App() {
  return (
    <AuthProvider>
      <BrowserRouter>
        <Routes>
          <Route path="/login" element={<GuestOnly><LoginPage /></GuestOnly>} />
          <Route path="/" element={<Protected><NotesPage /></Protected>} />
          <Route path="/archive" element={<Protected><NotesPage archived /></Protected>} />
          <Route path="*" element={<Navigate to="/" replace />} />
        </Routes>
      </BrowserRouter>
    </AuthProvider>
  );
}

import { useState } from 'react';
import { useAuth } from '../context/AuthContext';

export default function LoginPage() {
  const { login, register } = useAuth();
  const [mode, setMode] = useState('login');            // 'login' | 'register'
  const [form, setForm] = useState({ name: '', email: '', password: '' });
  const [error, setError] = useState(null);
  const [busy, setBusy] = useState(false);

  const onChange = (e) => setForm(f => ({ ...f, [e.target.name]: e.target.value }));

  async function onSubmit(e) {
    e.preventDefault();
    setBusy(true);
    setError(null);
    try {
      mode === 'login'
        ? await login({ email: form.email, password: form.password })
        : await register(form);
      // muvaffaqiyatli — GuestOnly avtomatik "/" ga yo'naltiradi
    } catch (err) {
      setError(err.details?.map(d => d.message).join('. ') || err.message);
    } finally {
      setBusy(false);
    }
  }

  return (
    <main className="auth">
      <form onSubmit={onSubmit} className="auth-card">
        <h1>🗒️ Eslatmalar</h1>

import { useState } from 'react';

export const COLORS = ['default', 'yellow', 'green', 'blue', 'pink', 'purple'];

// Yangi eslatma yaratish va mavjudini tahrirlash uchun bitta komponent
export default function NoteEditor({ initial, onSave, onCancel }) {
  const [title, setTitle] = useState(initial?.title ?? '');
  const [body, setBody] = useState(initial?.body ?? '');
  const [color, setColor] = useState(initial?.color ?? 'default');
  const [tagsText, setTagsText] = useState(initial?.tags?.join(', ') ?? '');
  const [error, setError] = useState(null);
  const [saving, setSaving] = useState(false);

  async function submit(e) {
    e.preventDefault();
    if (!title.trim() && !body.trim()) return setError('Sarlavha yoki matn yozing');
    const tags = tagsText.split(',').map(t => t.trim().toLowerCase()).filter(Boolean);
    setSaving(true);
    try {
      await onSave({ title, body, color, tags });
      if (!initial) { setTitle(''); setBody(''); setTagsText(''); setColor('default'); }
      setError(null);
    } catch (err) {
      setError(err.details?.map(d => `${d.field}: ${d.message}`).join('; ') || err.message);
    } finally {
      setSaving(false);
    }
  }

  return (
    <form className={`editor note-${color}`} onSubmit={submit}>
      <input placeholder="Sarlavha" value={title} onChange={e => setTitle(e.target.value)} maxLength={200} />
      <textarea placeholder="Eslatma matni..." rows={initial ? 8 : 3} value={body} onChange={e => setBody(e.target.value)} />
      <input placeholder="Teglar: ish, g'oyalar" value={tagsText} onChange={e => setTagsText(e.target.value)} />
      <div className="editor-bar">
        <div className="colors" role="radiogroup" aria-label="Rang">
          {COLORS.map(c => (
            <button key={c} type="button" role="radio" aria-checked={c === color} aria-label={c}
                    className={`swatch note-${c} ${c === color ? 'selected' : ''}`} onClick={() => setColor(c)} />
          ))}
        </div>
        {onCancel && <button type="button" onClick={onCancel}>Bekor</button>}
        <button className="primary" disabled={saving}>{saving ? 'Saqlanmoqda…' : 'Saqlash'}</button>
      </div>
      {error && <p className="error">{error}</p>}
    </form>
  );
}

import { useCallback, useEffect, useState } from 'react';
import { NavLink } from 'react-router';
import { useAuth } from '../context/AuthContext';
import { notesApi } from '../api';
import NoteCard from '../components/NoteCard';
import NoteEditor from '../components/NoteEditor';

function useDebounce(value, ms = 300) {
  const [v, setV] = useState(value);
  useEffect(() => { const t = setTimeout(() => setV(value), ms); return () => clearTimeout(t); }, [value, ms]);
  return v;
}

export default function NotesPage({ archived = false }) {
  const { user, logout } = useAuth();
  const [notes, setNotes] = useState([]);
  const [tags, setTags] = useState([]);
  const [status, setStatus] = useState('loading');      // 'loading' | 'ready' | 'error'
  const [q, setQ] = useState('');
  const [tag, setTag] = useState(null);
  const [editing, setEditing] = useState(null);
  const query = useDebounce(q);

  const loadTags = useCallback(() => notesApi.tags().then(setTags).catch(() => {}), []);

  // Filtr o'zgarganda ro'yxatni qayta yuklash; eski so'rovni bekor qilish
  useEffect(() => {
    const ctrl = new AbortController();
    setStatus('loading');
    const params = { archived: String(archived) };
    if (query) params.q = query;
    if (tag) params.tag = tag;
    notesApi.list(params, ctrl.signal)
      .then(({ data }) => { setNotes(data); setStatus('ready'); })
      .catch(err => { if (err.name !== 'AbortError') setStatus('error'); });
    return () => ctrl.abort();
  }, [archived, query, tag]);

  useEffect(() => { loadTags(); }, [loadTags]);

  const sortNotes = (list) =>
    [...list].sort((a, b) => (b.pinned - a.pinned) || b.updatedAt.localeCompare(a.updatedAt));

  async function create(data) {
    const note = await notesApi.create(data);
    setNotes(n => sortNotes([note, ...n]));
    loadTags();
  }

  // Optimistik yangilash: interfeys darhol o'zgaradi, xato bo'lsa — orqaga qaytariladi
  async function update(id, patch) {
    const prev = notes;
    setNotes(n => n.map(x => (x.id === id ? { ...x, ...patch } : x)));
    try {
      const saved = await notesApi.update(id, patch);
      setNotes(n => {
        const next = n.map(x => (x.id === id ? saved : x));
        // arxiv holati o'zgargan bo'lsa — joriy ro'yxatdan chiqaramiz
        return sortNotes(next.filter(x => x.archived === archived));
      });
      if ('tags' in patch || 'archived' in patch) loadTags();
    } catch (err) {
      setNotes(prev);
      alert(`Saqlanmadi: ${err.message}`);
    }
  }

  async function remove(note) {
    if (!confirm(`"${note.title || 'Sarlavhasiz'}" eslatmasi o'chirilsinmi?`)) return;
    const prev = notes;
    setNotes(n => n.filter(x => x.id !== note.id));
    try {
      await notesApi.remove(note.id);
      loadTags();
    } catch (err) {
      setNotes(prev);
      alert(err.message);
    }
  }

  async function saveEdit(data) {
    await update(editing.id, data);
    setEditing(null);
  }

  return (
    <div className="layout">
      <header className="topbar">
        <h1>🗒️ Eslatmalar</h1>
        <input type="search" placeholder="Qidirish..." value={q} onChange={e => setQ(e.target.value)} />
        <span className="user">{user.name}</span>
        <button onClick={logout}>Chiqish</button>
      </header>

      <aside className="sidebar">
        <NavLink to="/" end>📝 Eslatmalar</NavLink>
        <NavLink to="/archive">🗃️ Arxiv</NavLink>
        <h4>Teglar</h4>
        {tags.length === 0 && <small>Teg yo'q</small>}
        {tags.map(t => (
          <button key={t.name} className={t.name === tag ? 'active' : ''} onClick={() => setTag(t.name === tag ? null : t.name)}>
            #{t.name} <small>{t.count}</small>
          </button>
        ))}
      </aside>

      <main className="main">
        {!archived && <NoteEditor onSave={create} />}

        {tag && <p className="filter-info">Filtr: <b>#{tag}</b> <button onClick={() => setTag(null)}>✕</button></p>}
        {status === 'loading' && notes.length === 0 && <p className="muted">Yuklanmoqda...</p>}
        {status === 'error' && <p className="error">Eslatmalarni yuklab bo'lmadi. Sahifani yangilang.</p>}
        {status === 'ready' && notes.length === 0 && (
          <p className="muted">{query || tag ? 'Hech narsa topilmadi.' : archived ? 'Arxiv bo\'sh.' : 'Birinchi eslatmangizni yozing ✍️'}</p>
        )}

        <section className="grid">
          {notes.map(note => (
            <NoteCard key={note.id} note={note} onEdit={setEditing} onUpdate={update} onDelete={remove} onTagClick={setTag} />
          ))}
        </section>
      </main>

      {editing && (
        <dialog open className="modal" onKeyDown={e => e.key === 'Escape' && setEditing(null)}>
          <NoteEditor initial={editing} onSave={saveEdit} onCancel={() => setEditing(null)} />
        </dialog>
      )}
    </div>
  );
}
