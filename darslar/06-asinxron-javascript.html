Asinxron JavaScript va Fetch API

Event loop, callback, Promise, async/await, fetch bilan API'ga so'rov yuborish, xatolarni boshqarish, parallel so'rovlar va so'rovni bekor qilish.

O'rta · ~120 daqiqa

Veb-ilova doimo "kutadi": serverdan javobni, foydalanuvchi bosishini, faylning o'qilishini, taymerni. Agar har bir kutish butun dasturni to'xtatib qo'ysa, sahifa qotib qolardi. Asinxron dasturlash — uzoq davom etadigan amallarni boshlab, natijasini kutayotganda boshqa ishlarni davom ettirish san'ati. Frontend'ni backend bilan bog'lash aynan shu bilimga tayanadi.

#Event loop — JavaScript qanday ishlaydi

JavaScript bir oqimli (single-threaded): bir vaqtda faqat bitta kod qatorini bajaradi. Lekin brauzer (yoki Node.js) unga yordamchi mexanizmlar beradi:

  • Call Stack (chaqiruvlar steki) — hozir bajarilayotgan funksiyalar.
  • Web API'lar — brauzerning o'zi bajaradigan ishlar: taymerlar, tarmoq so'rovlari, hodisalar. Ular JS oqimini band qilmaydi.
  • Macrotask navbati — setTimeout, setInterval, hodisa ishlovchilari callback'lari.
  • Microtask navbati — Promise callback'lari (.then, await dan keyingi kod). Ustuvorligi yuqori.

#Callback'lar va "callback hell"

Asinxronlikning eng qadimiy usuli — natija tayyor bo'lganda chaqiriladigan funksiya (callback) berish:

JavaScript
function getUser(id, callback) {
  setTimeout(() => callback(null, { id, name: 'Ali' }), 500);
}

// Ketma-ket bog'liq amallar → "piramida" (callback hell)
getUser(1, (err, user) => {
  if (err) return console.error(err);
  getOrders(user.id, (err, orders) => {
    if (err) return console.error(err);
    getOrderDetails(orders[0].id, (err, details) => {
      if (err) return console.error(err);
      getShipping(details.shippingId, (err, shipping) => {
        // ... va hokazo — o'qish, xatolarni boshqarish juda qiyin
      });
    });
  });
});

#Promise

Promise (va'da) — kelajakda tayyor bo'ladigan natijani ifodalovchi obyekt. U uch holatdan birida bo'ladi:

  • pending — kutilmoqda;
  • fulfilled — muvaffaqiyatli bajarildi (qiymat bilan);
  • rejected — xato bilan tugadi (sabab bilan).

Holat bir marta o'zgaradi va boshqa o'zgarmaydi ("settled").

JavaScript
// Promise yaratish
function kutish(ms) {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

function tangaTashlash() {
  return new Promise((resolve, reject) => {
    setTimeout(() => {
      Math.random() > 0.5
        ? resolve('Gerb 🦅')
        : reject(new Error('Raqam chiqdi 🔢'));
    }, 1000);
  });
}

// Promise'ni ishlatish
tangaTashlash()
  .then((natija) => console.log('Yutdingiz:', natija))
  .catch((xato) => console.log('Yutqazdingiz:', xato.message))
  .finally(() => console.log('O\'yin tugadi'));

Zanjir (chaining)

.then() har doim yangi Promise qaytaradi. Callback'dan qaytarilgan qiymat keyingi .then ga o'tadi; agar Promise qaytarilsa — u tugashini kutadi. Bu callback piramidasini tekis zanjirga aylantiradi:

JavaScript
getUser(1)
  .then(user => getOrders(user.id))
  .then(orders => getOrderDetails(orders[0].id))
  .then(details => getShipping(details.shippingId))
  .then(shipping => console.log(shipping))
  .catch(err => console.error('Zanjirning istalgan joyidagi xato shu yerga keladi:', err));

#async / await

async/await — Promise ustidagi "sintaktik shakar": asinxron kodni xuddi sinxron koddek yuqoridan pastga o'qiladigan qiladi.

JavaScript
async function buyurtmaMalumoti(userId) {
  try {
    const user = await getUser(userId);          // Promise tugashini KUTADI (lekin oqimni bloklamaydi)
    const orders = await getOrders(user.id);
    const details = await getOrderDetails(orders[0].id);
    return details;                              // async funksiya DOIM Promise qaytaradi
  } catch (err) {
    console.error('Xato:', err.message);
    throw err;                                   // yuqoriga uzatish
  }
}

buyurtmaMalumoti(1).then(d => console.log(d));

// const d = await buyurtmaMalumoti(1);
  • await faqat async funksiya ichida (yoki ES modulning yuqori darajasida) ishlatiladi.
  • await funksiyani "pauza"ga qo'yadi, lekin event loop boshqa ishlarni davom ettiradi.
  • Xatolar oddiy try...catch bilan ushlanadi.

Ketma-ket va parallel bajarish

JavaScript
// ❌ SEKIN: bir-biriga bog'liq emas, lekin ketma-ket kutilyapti (1s + 1s + 1s = 3s)
const users = await fetchUsers();
const products = await fetchProducts();
const stats = await fetchStats();

// ✅ TEZ: parallel (eng uzuni = 1s)
const [users2, products2, stats2] = await Promise.all([
  fetchUsers(),
  fetchProducts(),
  fetchStats(),
]);

// ⚠️ forEach ichida await ishlamaydi (kutmaydi)!
ids.forEach(async id => await save(id));        // ❌ — hammasi "uchib ketadi"
for (const id of ids) await save(id);           // ✅ ketma-ket
await Promise.all(ids.map(id => save(id)));     // ✅ parallel

Qayta ishlatiladigan API yordamchisi

api.js
export class ApiError extends Error {
  constructor(status, message, data) {
    super(message);
    this.name = 'ApiError';
    this.status = status;
    this.data = data;
  }
}

const BASE_URL = 'http://localhost:3000/api';

export async function api(path, { method = 'GET', body, headers, ...rest } = {}) {
  const token = localStorage.getItem('token');
  const res = await fetch(BASE_URL + path, {
    method,
    headers: {
      ...(body && { 'Content-Type': 'application/json' }),
      ...(token && { Authorization: `Bearer ${token}` }),
      ...headers,
    },
    body: body ? JSON.stringify(body) : undefined,
    ...rest,
  });

  // 204 No Content — tana yo'q
  const data = res.status === 204 ? null : await res.json().catch(() => null);

  if (!res.ok) {
    throw new ApiError(res.status, data?.message || res.statusText, data);
  }
  return data;
}

CORS haqida qisqacha

Brauzer xavfsizlik uchun Same-Origin Policy ni qo'llaydi: http://localhost:5173 dagi sahifa http://localhost:3000 dagi API'ning javobini o'qiy olmaydi (port farq qilsa ham — boshqa "origin"). Ruxsat berish server zimmasida: u Access-Control-Allow-Origin sarlavhasini qaytarishi kerak. Konsolda "blocked by CORS policy" xatosini ko'rsangiz, frontendni emas, backendni sozlang (9-darsda cors middleware).

const BASE = 'https://jsonplaceholder.typicode.com';

async function loadDashboard() {
  console.time('dashboard');

  // To'rtta mustaqil so'rov PARALLEL; bittasi xato bo'lsa ham qolganlari ko'rsatiladi
  const results = await Promise.allSettled([
    getJSON('/users'),
    getJSON('/posts'),
    getJSON('/comments'),
    getJSON('/this-does-not-exist'),   // ataylab xato

  const [users, posts, comments, broken] = results.map(r =>

  results
    .filter(r => r.status === 'rejected')
    .forEach(r => console.warn('⚠️ Vidjet yuklanmadi:', r.reason.message));

  if (users && posts) {
    // Eng faol 3 ta muallif

      acc[p.userId] = (acc[p.userId] ?? 0) + 1;

Misol 4

Ishonchli so'rov: timeout, qayta urinish (retry) va eksponensial kutish

Mobil internet uzilib qolishi yoki server vaqtincha band (503) bo'lishi mumkin. Professional ilovalar bunday holatlarda so'rovni bir necha marta, har safar uzoqroq kutib, takrorlaydi.

fetchWithRetry.js
const sleep = (ms) => new Promise(r => setTimeout(r, ms));

const RETRYABLE = new Set([408, 429, 500, 502, 503, 504]);

export async function fetchWithRetry(url, options = {}, {
  retries = 3,
  timeout = 8000,
  baseDelay = 500,
  onRetry = () => {},
} = {}) {
  for (let attempt = 0; ; attempt++) {
    try {
      const res = await fetch(url, { ...options, signal: AbortSignal.timeout(timeout) });

      if (res.ok) return res;

      // 4xx xatolarni (404, 401...) takrorlashdan foyda yo'q
      if (!RETRYABLE.has(res.status) || attempt >= retries) {
        throw Object.assign(new Error(`HTTP ${res.status}`), { status: res.status });
      }
      // Server "Retry-After" sarlavhasini yuborgan bo'lsa, unga amal qilamiz
      const retryAfter = Number(res.headers.get('Retry-After')) * 1000;
      await wait(attempt, retryAfter);
    } catch (err) {
      const network = err.name === 'TypeError' || err.name === 'TimeoutError';
      if (!network || attempt >= retries) throw err;
      await wait(attempt);
    }
  }

  async function wait(attempt, forced) {
    // 500ms, 1000ms, 2000ms ... + tasodifiy "jitter" (hamma bir vaqtda qayta urinmasligi uchun)
    const delay = forced || baseDelay * 2 ** attempt + Math.random() * 200;
    onRetry({ attempt: attempt + 1, delay: Math.round(delay) });
    await sleep(delay);
  }
}

// Foydalanish
try {
  const res = await fetchWithRetry('https://httpbin.org/status/503', {}, {
    retries: 3,
    onRetry: ({ attempt, delay }) => console.log(`${attempt}-qayta urinish, ${delay}ms kutamiz...`),
  });
  console.log(await res.text());
} catch (err) {
  console.error('Barcha urinishlar muvaffaqiyatsiz:', err.message);
}
Faqat idempotent so'rovlarni (GET, PUT, DELETE) avtomatik takrorlang. POST /payments ni qayta yuborish ikki marta to'lovga olib kelishi mumkin (1-darsdagi idempotentlik tushunchasini eslang). Bunday holatlar uchun serverlar "Idempotency-Key" sarlavhasidan foydalanadi.
