Express.js va REST API

Express asoslari, marshrutlash, middleware zanjiri, REST API dizayni, validatsiya, xatolarni markazlashgan boshqarish, CORS va loyiha tuzilmasi (MVC).

O'rta · ~140 daqiqa

Express — Node.js uchun eng mashhur minimalistik web-freymvork. U oldingi darsda qo'lda yozgan narsalarimizni — marshrutlash, JSON'ni o'qish, statik fayllar, xatolarni boshqarish — oddiy va tushunarli API bilan beradi. Bu darsda frontend ilovalar "gaplashadigan" haqiqiy REST API qurishni o'rganamiz.

#Marshrutlash (Routing)

Marshrut = HTTP metod + yo'l + ishlovchi funksiya(lar).

JavaScript
app.get('/api/books', handler);          // o'qish
app.post('/api/books', handler);         // yaratish
app.put('/api/books/:id', handler);      // to'liq almashtirish
app.patch('/api/books/:id', handler);    // qisman yangilash
app.delete('/api/books/:id', handler);   // o'chirish
app.all('/api/secret', handler);         // istalgan metod

// Bitta yo'l uchun bir nechta metod
app.route('/api/books/:id')
  .get(getBook)
  .patch(updateBook)
  .delete(deleteBook);

So'rov (req) obyektidan ma'lumot olish

JavaScript
// GET /api/users/42/orders?status=paid&page=2
app.get('/api/users/:userId/orders', (req, res) => {
  req.params;          // { userId: '42' }               — yo'l parametrlari (doim satr!)
  req.query;           // { status: 'paid', page: '2' }  — query parametrlar (doim satr!)
  req.body;            // POST/PUT tanasi (express.json() kerak)
  req.headers;         // barcha sarlavhalar (kichik harfda)
  req.get('authorization');
  req.method;          // 'GET'
  req.path;            // '/api/users/42/orders'
  req.ip;              // mijoz IP manzili
  req.cookies;         // cookie-parser middleware bilan
});

Har bir middleware so'rovni o'zgartirishi, javob qaytarib zanjirni to'xtatishi yoki next() bilan keyingisiga uzatishi mumkin
JavaScript
function middleware(req, res, next) {
  // 1. req/res ni o'qish yoki o'zgartirish
  // 2. va keyin:
  next();              // → keyingi middleware'ga o'tish
  // yoki: res.status(401).json(...)  — zanjirni to'xtatish
  // yoki: next(err)                  — xato ishlovchisiga o'tish
}

Middleware turlari

JavaScript
// 1. Ilova darajasida — barcha so'rovlar uchun
app.use(express.json({ limit: '1mb' }));             // JSON tana

#REST API dizayni

REST (Representational State Transfer) — API'ni loyihalash uslubi. Asosiy g'oya: hamma narsa — resurs (kitob, foydalanuvchi, buyurtma), har bir resurs URL bilan aniqlanadi, ular ustidagi amallar HTTP metodlari bilan ifodalanadi.

AmalMetod va URLMuvaffaqiyatli javob
Ro'yxatGET /api/books200 + massiv
Bittasini olishGET /api/books/42200 + obyekt / 404
YaratishPOST /api/books201 + yaratilgan obyekt
To'liq yangilashPUT /api/books/42200
Qisman yangilashPATCH /api/books/42200
O'chirishDELETE /api/books/42204
Ichki resursGET /api/authors/7/books200

Yaxshi API qoidalari

  • URL'da otlar, ko'plikda: /api/books ✅, /api/getBooks ❌, /api/book/delete/5 ❌.
  • Kichik harflar va defis: /api/order-items.
  • Filtrlash, saralash, sahifalash — query orqali: /api/books?genre=roman&sort=-price&page=2&limit=20.
  • To'g'ri status kodlar (1-dars jadvali).
  • Xatolar bir xil formatda: { "error": { "message": "...", "details": [...] } }.
  • Versiyalash: /api/v1/books — API'ni buzuvchi o'zgarishlar kiritilganda eski mijozlar ishlashda davom etadi.
  • Sahifalangan javobda meta-ma'lumot: { data: [...], meta: { page, limit, total } }.

#Xatolarni markazlashgan boshqarish

Har bir marshrutda try/catch yozib, xatoni qo'lda formatlash — takrorlanuvchi kod. To'g'ri yondashuv: marshrutlar xatoni tashlaydi (throw), bitta xato middleware'i ularni bir xil formatda javobga aylantiradi.

src/utils/errors.js
export class HttpError extends Error {
  constructor(status, message, details) {
    super(message);
    this.status = status;
    this.details = details;
  }
  static badRequest(msg = 'Noto\'g\'ri so\'rov', details) { return new HttpError(400, msg, details); }
  static unauthorized(msg = 'Avtorizatsiya talab qilinadi') { return new HttpError(401, msg); }
  static forbidden(msg = 'Ruxsat yo\'q') { return new HttpError(403, msg); }
  static notFound(msg = 'Topilmadi') { return new HttpError(404, msg); }
  static conflict(msg) { return new HttpError(409, msg); }
}
src/middleware/errorHandler.js
import { HttpError } from '../utils/errors.js';

export function notFound(req, res, next) {
  next(HttpError.notFound(`Marshrut topilmadi: ${req.method} ${req.originalUrl}`));
}

// eslint-disable-next-line no-unused-vars
export function errorHandler(err, req, res, next) {
  // Noto'g'ri JSON (express.json() tashlaydi)
  if (err.type === 'entity.parse.failed') {
    err = HttpError.badRequest('So\'rov tanasidagi JSON noto\'g\'ri');
  }

  const status = err.status ?? 500;
  if (status >= 500) console.error(err);   // kutilmagan xatolarni log qilamiz

  res.status(status).json({
    error: {
      message: status >= 500 && process.env.NODE_ENV === 'production'
        ? 'Ichki server xatosi'           // 🔒 production'da ichki tafsilotlarni oshkor qilmaymiz
        : err.message,
      ...(err.details && { details: err.details }),
    },
  });
}

import { HttpError } from '../utils/errors.js';

// Sxema bo'yicha req.body / req.query ni tekshiruvchi middleware "fabrikasi"
export const validate = (schema, source = 'body') => (req, res, next) => {
  const result = schema.safeParse(req[source]);
  if (!result.success) {
    const details = result.error.issues.map(i => ({ field: i.path.join('.'), message: i.message }));
    return next(HttpError.badRequest('Validatsiya xatosi', details));
  }
  req.validated = { ...req.validated, [source]: result.data };
  next();
};
src/services/books.service.js
import { JsonStore } from '../db/jsonStore.js';
import { HttpError } from '../utils/errors.js';

const store = new JsonStore('data/books.json');

const SORTERS = {
  price: (a, b) => a.price - b.price,
  '-price': (a, b) => b.price - a.price,
  title: (a, b) => a.title.localeCompare(b.title),
  '-createdAt': (a, b) => b.createdAt.localeCompare(a.createdAt),
};

export async function listBooks({ q, genre, minPrice, maxPrice, sort, page, limit }) {
  const needle = q?.toLowerCase();
  const all = await store.all({
    filter: (b) =>
      (!needle || b.title.toLowerCase().includes(needle) || b.author.toLowerCase().includes(needle)) &&
      (!genre || b.genre === genre) &&
      (minPrice === undefined || b.price >= minPrice) &&
      (maxPrice === undefined || b.price <= maxPrice),
    sort: SORTERS[sort],
  });
  const start = (page - 1) * limit;
  return {
    data: all.slice(start, start + limit),
    meta: { page, limit, total: all.length, pages: Math.ceil(all.length / limit) },
  };
}

export async function getBook(id) {
  const book = await store.get(id);
  if (!book) throw HttpError.notFound('Kitob topilmadi');
  return book;
}

export async function createBook(data) {
  const duplicate = await store.all({
    filter: b => b.title.toLowerCase() === data.title.toLowerCase() && b.author === data.author,
  });
  if (duplicate.length) throw HttpError.conflict('Bu kitob allaqachon mavjud');
  return store.create(data);
}

export async function updateBook(id, patch) {
  const book = await store.update(id, patch);
  if (!book) throw HttpError.notFound('Kitob topilmadi');
  return book;
}

export async function deleteBook(id) {
  if (!(await store.remove(id))) throw HttpError.notFound('Kitob topilmadi');
}

src/middleware/common.js
import { randomUUID } from 'node:crypto';

// 1. Har bir so'rovga noyob ID — loglarda bitta so'rovning izini kuzatish uchun
export function requestId(req, res, next) {
  req.id = req.get('X-Request-Id') ?? randomUUID();
  res.set('X-Request-Id', req.id);
  next();
}

// 2. Tuzilgan (structured) logger: javob yuborilgandan keyin yozadi
export function logger(req, res, next) {
  const start = process.hrtime.bigint();
  res.on('finish', () => {
    const ms = Number(process.hrtime.bigint() - start) / 1e6;
    const line = {
      time: new Date().toISOString(),
      id: req.id,
      method: req.method,
      url: req.originalUrl,
      status: res.statusCode,
      ms: Math.round(ms * 10) / 10,
      ip: req.ip,
    };
    (res.statusCode >= 500 ? console.error : console.log)(JSON.stringify(line));
  });
  next();
}

// 3. Sekin so'rovlarni aniqlash
export const slowWarning = (thresholdMs = 1000) => (req, res, next) => {
  const t = setTimeout(() => console.warn(`🐢 Sekin so'rov (${thresholdMs}ms+): ${req.method} ${req.originalUrl}`), thresholdMs);
  res.on('finish', () => clearTimeout(t));
  next();
};

// 4. API kaliti bilan himoya (masalan, ichki xizmatlar uchun)
export const requireApiKey = (req, res, next) => {
  const key = req.get('X-API-Key');
  if (!key || key !== process.env.INTERNAL_API_KEY) {
    return res.status(401).json({ error: { message: 'API kaliti noto\'g\'ri' } });
  }
  next();
};

// 5. Juda oddiy xotiradagi rate limiter (o'rganish uchun; productionda express-rate-limit + Redis)
export function rateLimit({ windowMs = 60_000, max = 60 } = {}) {
  const hits = new Map();   // ip → { count, reset }
  return (req, res, next) => {
    const now = Date.now();
    const entry = hits.get(req.ip) ?? { count: 0, reset: now + windowMs };
    if (now > entry.reset) Object.assign(entry, { count: 0, reset: now + windowMs });
    entry.count++;
    hits.set(req.ip, entry);

    res.set('RateLimit-Remaining', Math.max(0, max - entry.count));
    if (entry.count > max) {
      res.set('Retry-After', Math.ceil((entry.reset - now) / 1000));
      return res.status(429).json({ error: { message: 'Juda ko\'p so\'rov. Birozdan keyin urinib ko\'ring.' } });
    }
    next();
  };
}

const ALLOWED = new Map([
  ['image/jpeg', '.jpg'],
  ['image/png', '.png'],
  ['image/webp', '.webp'],
]);

const upload = multer({
  storage: multer.diskStorage({
    destination: 'uploads/covers',
    // 🔒 Foydalanuvchi bergan nomni ISHLATMAYMIZ — tasodifiy nom + ishonchli kengaytma
    filename: (req, file, cb) => cb(null, randomUUID() + ALLOWED.get(file.mimetype)),
  }),
  limits: { fileSize: 2 * 1024 * 1024, files: 1 },    // 2 MB
  fileFilter: (req, file, cb) =>
    ALLOWED.has(file.mimetype)
      ? cb(null, true)
      : cb(HttpError.badRequest('Faqat JPG, PNG yoki WEBP rasm')),
});

const router = Router();

router.post('/books/:id/cover', upload.single('cover'), async (req, res) => {
  if (!req.file) throw HttpError.badRequest('Rasm yuborilmadi (maydon nomi: cover)');
  const url = `/uploads/covers/${req.file.filename}`;
  // await booksService.updateBook(req.params.id, { coverUrl: url });
  res.status(201).json({ url, size: req.file.size });
});

    const API = '/api/v1/books';           // bir xil origin — CORS kerak emas
    const list = document.querySelector('#list');
    const msg = document.querySelector('#msg');

    async function request(url, options) {
      const res = await fetch(url, {
        ...options,
        headers: { 'Content-Type': 'application/json' },
      });
      if (res.status === 204) return null;
      const data = await res.json();
      if (!res.ok) {
        const details = data.error.details?.map(d => `${d.field}: ${d.message}`).join('; ');
        throw new Error(details || data.error.message);
      }
      return data;
    }

    async function load(q = '') {
      const { data, meta } = await request(`${API}?limit=50&q=${encodeURIComponent(q)}`);
      list.replaceChildren(...data.map(b => {
        const row = document.createElement('div');
        row.className = 'book';
        const info = document.createElement('span');
        info.textContent = `${b.title} — ${b.author} (${b.price.toLocaleString()} so'm)`;
        const del = document.createElement('button');
        del.textContent = '🗑';
        del.onclick = async () => { await request(`${API}/${b.id}`, { method: 'DELETE' }); load(q); };
        row.append(info, del);
        return row;
      }));
      if (!meta.total) list.textContent = 'Kitob topilmadi';
    }

    document.querySelector('#add').addEventListener('submit', async (e) => {
      e.preventDefault();
      const body = Object.fromEntries(new FormData(e.target));
      body.price = Number(body.price);
      try {
        await request(API, { method: 'POST', body: JSON.stringify(body) });
        e.target.reset();
