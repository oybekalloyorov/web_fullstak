REST API dizayni

REST (Representational State Transfer) — API'ni loyihalash uslubi. Asosiy g'oya: hamma narsa — resurs (kitob, foydalanuvchi, buyurtma), har bir resurs URL bilan aniqlanadi, ular ustidagi amallar HTTP metodlari bilan ifodalanadi.

YaratishPOST /api/books201 + yaratilgan obyekt
To'liq yangilashPUT /api/books/42200
Qisman yangilashPATCH /api/books/42200

Yaxshi API qoidalari

  • URL'da otlar, ko'plikda: /api/books ✅, /api/getBooks ❌, /api/book/delete/5 ❌.
  • Kichik harflar va defis: /api/order-items.
  • Filtrlash, saralash, sahifalash — query orqali: /api/books?genre=roman&sort=-price&page=2&limit=20.
  • To'g'ri status kodlar (1-dars jadvali).
  • Xatolar bir xil formatda: { "error": { "message": "...", "details": [...] } }.
  • Versiyalash: /api/v1/books — API'ni buzuvchi o'zgarishlar kiritilganda eski mijozlar ishlashda davom etadi.
  • Sahifalangan javobda meta-ma'lumot: { data: [...], meta: { page, limit, total } }.

Xatolarni markazlashgan boshqarish

Har bir marshrutda try/catch yozib, xatoni qo'lda formatlash — takrorlanuvchi kod. To'g'ri yondashuv: marshrutlar xatoni tashlaydi (throw), bitta xato middleware'i ularni bir xil formatda javobga aylantiradi.

export class HttpError extends Error {
  constructor(status, message, details) {
    super(message);
    this.status = status;
    this.details = details;
  }
  static badRequest(msg = 'Noto\'g\'ri so\'rov', details) { return new HttpError(400, msg, details); }
  static unauthorized(msg = 'Avtorizatsiya talab qilinadi') { return new HttpError(401, msg); }
  static forbidden(msg = 'Ruxsat yo\'q') { return new HttpError(403, msg); }
  static notFound(msg = 'Topilmadi') { return new HttpError(404, msg); }
  static conflict(msg) { return new HttpError(409, msg); }
}
import { HttpError } from '../utils/errors.js';

export function notFound(req, res, next) {
  next(HttpError.notFound(`Marshrut topilmadi: ${req.method} ${req.originalUrl}`));
}

// eslint-disable-next-line no-unused-vars
export function errorHandler(err, req, res, next) {
  // Noto'g'ri JSON (express.json() tashlaydi)
  if (err.type === 'entity.parse.failed') {
    err = HttpError.badRequest('So\'rov tanasidagi JSON noto\'g\'ri');
  }

  const status = err.status ?? 500;
  if (status >= 500) console.error(err);   // kutilmagan xatolarni log qilamiz

  res.status(status).json({
    error: {
      message: status >= 500 && process.env.NODE_ENV === 'production'
        ? 'Ichki server xatosi'           // 🔒 production'da ichki tafsilotlarni oshkor qilmaymiz
        : err.message,
      ...(err.details && { details: err.details }),
    },
  });
}

export async function listBooks({ q, genre, minPrice, maxPrice, sort, page, limit }) {
  const needle = q?.toLowerCase();
  const all = await store.all({
    filter: (b) =>
      (!needle || b.title.toLowerCase().includes(needle) || b.author.toLowerCase().includes(needle)) &&
      (!genre || b.genre === genre) &&
      (minPrice === undefined || b.price >= minPrice) &&
      (maxPrice === undefined || b.price <= maxPrice),
    sort: SORTERS[sort],
  });
  const start = (page - 1) * limit;
  return {
    data: all.slice(start, start + limit),
    meta: { page, limit, total: all.length, pages: Math.ceil(all.length / limit) },
  };
}

  next();

// 2. Tuzilgan (structured) logger: javob yuborilgandan keyin yozadi
export function logger(req, res, next) {
  const start = process.hrtime.bigint();
  res.on('finish', () => {
    const ms = Number(process.hrtime.bigint() - start) / 1e6;
    const line = {
      time: new Date().toISOString(),
      id: req.id,
      method: req.method,
      url: req.originalUrl,
      status: res.statusCode,
      ms: Math.round(ms * 10) / 10,
      ip: req.ip,
    };
    (res.statusCode >= 500 ? console.error : console.log)(JSON.stringify(line));
  });
  next();
}

// 5. Juda oddiy xotiradagi rate limiter (o'rganish uchun; productionda express-rate-limit + Redis)
export function rateLimit({ windowMs = 60_000, max = 60 } = {}) {
  const hits = new Map();   // ip → { count, reset }
  return (req, res, next) => {
    const now = Date.now();
    const entry = hits.get(req.ip) ?? { count: 0, reset: now + windowMs };
    if (now > entry.reset) Object.assign(entry, { count: 0, reset: now + windowMs });
    entry.count++;
    hits.set(req.ip, entry);

    res.set('RateLimit-Remaining', Math.max(0, max - entry.count));
    if (entry.count > max) {
      res.set('Retry-After', Math.ceil((entry.reset - now) / 1000));
      return res.status(429).json({ error: { message: 'Juda ko\'p so\'rov. Birozdan keyin urinib ko\'ring.' } });
    }
    next();
  };
}
