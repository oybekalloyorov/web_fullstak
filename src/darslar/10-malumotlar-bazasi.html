Oldingi darsda ma'lumotlarni JSON faylda saqladik. Bu yondashuv ming foydalanuvchida "yiqiladi": qidiruv sekin, bir vaqtdagi yozuvlar to'qnashadi, "shu muallifning 2020-yildan keyingi, narxi 50 mingdan arzon kitoblari" kabi so'rovlarni har safar qo'lda yozish kerak. Normalizatsiya — ma'lumotlarni takrorlanmaydigan va ziddiyatsiz qilib jadvallarga bo'lish. Yomon dizayn:

customers(id, name, phone)
orders(id, customer_id → customers, status, total, created_at)
order_items(order_id → orders, book_id → books, qty, price)   -- narx sotilgan paytdagi qiymat!

Asosiy qoidalar (1NF–3NF): har bir katakda bitta qiymat (ro'yxat emas); har bir ustun to'liq birlamchi kalitga bog'liq; kalit bo'lmagan ustunlar bir-biriga bog'liq emas. Amaliy qoida: "bitta fakt — bitta joyda".

order_items.price ataylab takrorlangan: kitob narxi keyin o'zgarsa ham, eski buyurtmada sotilgan narx saqlanib qolishi kerak. Bu — ongli denormalizatsiya.

Indekslar

Indekssiz baza WHERE email = '...' so'rovi uchun har bir qatorni tekshiradi (Seq Scan). Indeks — kitob oxiridagi alifbo ko'rsatkichi kabi: kerakli joyni darhol topadi (B-tree, O(log n)).

CREATE INDEX idx_books_author ON books(author_id);         -- JOIN va filtrlar uchun
CREATE UNIQUE INDEX idx_users_email ON users(lower(email)); -- katta-kichik harfsiz noyoblik
CREATE INDEX idx_orders_user_date ON orders(user_id, created_at DESC); -- kompozit

-- So'rov rejasini ko'rish
EXPLAIN ANALYZE SELECT * FROM books WHERE author_id = 5;
-- Index Scan using idx_books_author ... (actual time=0.02..0.03 ms)
  • PRIMARY KEY va UNIQUE avtomatik indekslanadi; tashqi kalitlar — yo'q, ularni qo'lda indekslang.
  • Indeks o'qishni tezlashtiradi, lekin yozishni biroz sekinlashtiradi va joy egallaydi — faqat haqiqatan so'raladigan ustunlarga qo'ying.

Tranzaksiyalar va ACID

Tranzaksiya — bir nechta amalni "hammasi yoki hech biri" tamoyili bilan bajarish. Klassik misol — pul o'tkazmasi: birinchi hisobdan yechildi, ikkinchisiga yozishda server o'chib qoldi — pul "yo'qoladi". Tranzaksiya buni oldini oladi.

BEGIN;
  UPDATE accounts SET balance = balance - 500000 WHERE id = 1;
  UPDATE accounts SET balance = balance + 500000 WHERE id = 2;
COMMIT;      -- ikkalasi birga saqlanadi
-- yoki xato bo'lsa: ROLLBACK;  — hech narsa o'zgarmaydi
  • Atomicity — bo'linmaslik: hammasi yoki hech biri.
  • Consistency — izchillik: cheklovlar (CHECK balance >= 0) har doim bajariladi.
  • Isolation — ajratilganlik: parallel tranzaksiyalar bir-birining yarim natijasini ko'rmaydi.
  • Durability — barqarorlik: COMMIT dan keyin ma'lumot, hatto elektr o'chsa ham, saqlanadi.

SELECT p.id, p.title, p.slug, p.published_at,
       u.username AS author,
       (SELECT COUNT(*) FROM comments c WHERE c.post_id = p.id) AS comments,
       (SELECT COUNT(*) FROM likes l WHERE l.post_id = p.id)    AS likes,
       COALESCE(array_agg(t.name ORDER BY t.name) FILTER (WHERE t.id IS NOT NULL), '{}') AS tags
FROM posts p
JOIN users u          ON u.id = p.author_id
LEFT JOIN post_tags pt ON pt.post_id = p.id
LEFT JOIN tags t       ON t.id = pt.tag_id
WHERE p.status = 'published'
GROUP BY p.id, u.username
ORDER BY p.published_at DESC
LIMIT 10;

  UNION ALL
  SELECT author_id, 'comment' FROM comments WHERE created_at > now() - interval '30 days'

       COUNT(*) FILTER (WHERE kind = 'post')    AS posts,

import { query } from '../db/pool.js';
import { HttpError } from '../utils/errors.js';

const SORT_SQL = {                    // whitelist — dinamik ORDER BY uchun xavfsiz usul
  price: 'price ASC',
  '-price': 'price DESC',
  title: 'title ASC',
  '-createdAt': 'created_at DESC',
};

const COLUMNS = `id, title, author, price, stock, genre, year,
                 created_at AS "createdAt", updated_at AS "updatedAt"`;

export async function listBooks({ q, genre, minPrice, maxPrice, sort, page, limit }) {
  const where = [];
  const params = [];
  const add = (sql, value) => { params.push(value); where.push(sql.replaceAll('?', `$${params.length}`)); };

  if (q) add('(title ILIKE ? OR author ILIKE ?)', `%${q}%`);   // ikkala ? → bitta $n
  if (genre) add('genre = ?', genre);
  if (minPrice !== undefined) add('price >= ?', minPrice);
  if (maxPrice !== undefined) add('price <= ?', maxPrice);

  const whereSql = where.length ? `WHERE ${where.join(' AND ')}` : '';
  const offset = (page - 1) * limit;

  const [{ rows }, { rows: [{ total }] }] = await Promise.all([
    query(
      `SELECT ${COLUMNS} FROM books ${whereSql}
       ORDER BY ${SORT_SQL[sort]} LIMIT $${params.length + 1} OFFSET $${params.length + 2}`,
      [...params, limit, offset]
    ),
    query(`SELECT COUNT(*)::int AS total FROM books ${whereSql}`, params),
  ]);

  return { data: rows, meta: { page, limit, total, pages: Math.ceil(total / limit) } };
}

export async function getBook(id) {
  const { rows: [book] } = await query(`SELECT ${COLUMNS} FROM books WHERE id = $1`, [id]);
  if (!book) throw HttpError.notFound('Kitob topilmadi');
  return book;
}

export async function createBook({ title, author, price, stock, genre, year }) {
  try {
    const { rows: [book] } = await query(
      `INSERT INTO books (title, author, price, stock, genre, year)
       VALUES ($1, $2, $3, $4, $5, $6) RETURNING ${COLUMNS}`,
      [title, author, price, stock, genre, year ?? null]
    );
    return book;
  } catch (err) {
    if (err.code === '23505') throw HttpError.conflict('Bu kitob allaqachon mavjud'); // unique_violation
    throw err;
  }
}

export async function updateBook(id, patch) {
  const allowed = ['title', 'author', 'price', 'stock', 'genre', 'year'];
  const entries = Object.entries(patch).filter(([k]) => allowed.includes(k));
  if (!entries.length) return getBook(id);

  const set = entries.map(([k], i) => `${k} = $${i + 2}`).join(', ');
  const { rows: [book] } = await query(
    `UPDATE books SET ${set}, updated_at = now() WHERE id = $1 RETURNING ${COLUMNS}`,
    [id, ...entries.map(([, v]) => v)]
  );
  if (!book) throw HttpError.notFound('Kitob topilmadi');
  return book;
}

export async function deleteBook(id) {
  const { rowCount } = await query('DELETE FROM books WHERE id = $1', [id]);
  if (!rowCount) throw HttpError.notFound('Kitob topilmadi');
}

add() yordamchisi qiymatni params ga qo'shib, shartdagi ? belgilarini tegishli $n ga almashtiradi (q shartida ikkala ? bitta parametrni ishlatadi). Ustun nomlari faqat allowed ro'yxatidan, qiymatlar esa faqat parametrlar orqali — injection'ga joy yo'q. UUID formatida bo'lmagan id kelsa, PostgreSQL 22P02 xatosini beradi — uni ham errorHandler da 404/400 ga aylantirish mumkin.

import { pool } from '../db/pool.js';
import { HttpError } from '../utils/errors.js';

// Qayta ishlatiladigan tranzaksiya yordamchisi
export async function withTransaction(fn) {
  const client = await pool.connect();     // tranzaksiya BITTA ulanishda bo'lishi shart
  try {
    await client.query('BEGIN');
    const result = await fn(client);
    await client.query('COMMIT');
    return result;
  } catch (err) {
    await client.query('ROLLBACK');
    throw err;
  } finally {
    client.release();                      // ulanishni pool'ga qaytarish — unutilsa pool "tugaydi"
  }
}

export function placeOrder(userId, items) {   // items: [{ bookId, qty }]
  return withTransaction(async (db) => {
    const ids = items.map(i => i.bookId);

    // FOR UPDATE — bu qatorlarni tranzaksiya tugaguncha qulflaydi:
    // ikki xaridor oxirgi nusxani bir vaqtda sotib ololmaydi
    const { rows: books } = await db.query(
      'SELECT id, title, price, stock FROM books WHERE id = ANY($1) ORDER BY id FOR UPDATE',
      [ids]
    );
    const byId = new Map(books.map(b => [b.id, b]));

    let total = 0;
    for (const { bookId, qty } of items) {
      const book = byId.get(bookId);
      if (!book) throw HttpError.notFound(`Kitob topilmadi: ${bookId}`);
      if (book.stock < qty) {
        throw HttpError.conflict(`"${book.title}" omborda yetarli emas (bor: ${book.stock}, so'raldi: ${qty})`);
      }
      total += book.price * qty;
    }

    const { rows: [order] } = await db.query(
      'INSERT INTO orders (user_id, total) VALUES ($1, $2) RETURNING *',
      [userId, total]
    );

    for (const { bookId, qty } of items) {
      await db.query('UPDATE books SET stock = stock - $1 WHERE id = $2', [qty, bookId]);
      await db.query(
        'INSERT INTO order_items (order_id, book_id, qty, price) VALUES ($1, $2, $3, $4)',
        [order.id, bookId, qty, byId.get(bookId).price]
      );
    }

    return { ...order, items };
  });
}

Uchta muhim nuqta: tranzaksiya uchun pool.query emas, bitta client kerak (aks holda har bir so'rov boshqa ulanishga tushishi mumkin); FOR UPDATE "race condition" ni yo'q qiladi; ORDER BY id bilan qulflash tartibi bir xil bo'lgani uchun "deadlock" ehtimoli kamayadi. Xato tashlansa — ROLLBACK, ombor qiymatlari o'zgarmaydi.
