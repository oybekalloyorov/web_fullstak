# 2-daraja: Generic view'lar
```python blog/api_views.py
from rest_framework import generics

class MaqolaListCreate(generics.ListCreateAPIView):          # GET ro'yxat + POST
    queryset = Maqola.objects.filter(chop_etilgan=True).select_related("kategoriya", "muallif")
    serializer_class = MaqolaSerializer

    def perform_create(self, serializer):
        serializer.save(muallif=self.request.user)           # muallifni qo'shamiz

class MaqolaDetail(generics.RetrieveUpdateDestroyAPIView):   # GET + PUT + PATCH + DELETE
    queryset = Maqola.objects.all()
    serializer_class = MaqolaSerializer
```
```python blog/urls.py
path("api/maqolalar/", api_views.MaqolaListCreate.as_view()),
path("api/maqolalar/<int:pk>/", api_views.MaqolaDetail.as_view()),   # generic'lar "pk" kutadi
```
! ~15 qatorda to'liq CRUD! Lekin hozir **istalgan** foydalanuvchi begona maqolani o'zgartira oladi...

# Ruxsatlar (permissions)
```python blog/permissions.py
from rest_framework import permissions

class MuallifYokiFaqatOqish(permissions.BasePermission):
    """O'qish — hammaga; o'zgartirish — faqat muallifga."""
    def has_object_permission(self, request, view, obj):
        if request.method in permissions.SAFE_METHODS:   # GET, HEAD, OPTIONS
            return True
        return obj.muallif == request.user
```
```python
permission_classes = [permissions.IsAuthenticatedOrReadOnly, MuallifYokiFaqatOqish]
```
| Tayyor ruxsat | Ma'nosi |
|---|---|
| `AllowAny` | Hammaga ochiq |
| `IsAuthenticated` | Faqat kirganlar |
| `IsAuthenticatedOrReadOnly` | O'qish — hammaga, yozish — kirganlarga |
| `IsAdminUser` | Faqat admin |
- Begona PUT/DELETE → **403 Forbidden** (38-mavzu qoidasi, API'da)

# 3-daraja: ViewSet
```python
from rest_framework import viewsets, permissions, filters

class MaqolaViewSet(viewsets.ModelViewSet):     # list, create, retrieve, update, destroy
    serializer_class = MaqolaSerializer
    permission_classes = [permissions.IsAuthenticatedOrReadOnly, MuallifYokiFaqatOqish]
    filter_backends = [filters.SearchFilter, filters.OrderingFilter]
    search_fields = ["sarlavha", "matn"]              # ?search=futbol
    ordering_fields = ["yaratilgan", "korishlar"]     # ?ordering=-korishlar

    def get_queryset(self):
        qs = Maqola.objects.filter(chop_etilgan=True).select_related("kategoriya", "muallif")
        kategoriya = self.request.query_params.get("kategoriya")   # ?kategoriya=sport
        if kategoriya:
            qs = qs.filter(kategoriya__slug=kategoriya)
        return qs

    def perform_create(self, serializer):
        serializer.save(muallif=self.request.user)
```

# Router — URL'lar avtomatik
```python config/urls.py
from rest_framework.routers import DefaultRouter
from blog.api_views import MaqolaViewSet, KategoriyaViewSet

router = DefaultRouter()
router.register("maqolalar", MaqolaViewSet, basename="maqola")
router.register("kategoriyalar", KategoriyaViewSet)

urlpatterns = [
    path("admin/", admin.site.urls),
    path("api/", include(router.urls)),
    path("api-auth/", include("rest_framework.urls")),
    path("", include("blog.urls")),
]
```
| URL | Metodlar | ViewSet metodi |
|---|---|---|
| `/api/maqolalar/` | GET, POST | list, create |
| `/api/maqolalar/5/` | GET, PUT, PATCH, DELETE | retrieve, update, partial_update, destroy |
! Oldingi qo'lda yozilgan `api/maqolalar/...` yo'llarini o'chiring — ikki joyda bo'lmasin

# Front-end: JavaScript mijoz (29-loyiha)
```js
const API = "http://127.0.0.1:8000/api/maqolalar/";

async function yukla(url) {
  holat.textContent = "Yuklanmoqda...";
  try {
    const javob = await fetch(url);
    if (!javob.ok) throw new Error(`Server xatosi: ${javob.status}`);
    const data = await javob.json();          // { count, next, previous, results }
    royxat.innerHTML = "";
    data.results.forEach(m => {
      const karta = document.createElement("article");
      karta.className = "karta";
      const h3 = document.createElement("h3");
      h3.textContent = m.sarlavha;            // textContent — XSS'dan himoya
      karta.append(h3);
      royxat.append(karta);
    });
    holat.textContent = `Jami: ${data.count} ta maqola`;
  } catch (err) {
    holat.textContent = `❌ ${err.message}. Django server ishlayaptimi?`;
  }
}
yukla(API);
```

# Ko'p uchraydigan xatolar
- ❌ `blocked by CORS policy` — corsheaders yo'q, middleware boshida emas yoki manzil (port ham!) ro'yxatda yo'q
- ❌ `get_queryset` bilan ViewSet'da router `basename` unutilgan
- ❌ Generic view URL'ida `<int:id>` — `pk` kerak
- ❌ `fetch` natijasini `.json()` siz yoki `await` siz ishlatish
