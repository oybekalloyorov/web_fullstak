# Funksiyalar: def
```python
def salomlash(ism):
    """Berilgan ism bilan salomlashadi."""     # docstring
    print(f"Assalomu alaykum, {ism}!")

salomlash("Madina")

def tortburchak_yuzi(eni, boyi):
    return eni * boyi

yuza = tortburchak_yuzi(5, 3)          # 15
```
| JavaScript | Python |
|---|---|
| `function qosh(a, b) { return a + b; }` | `def qosh(a, b): return a + b` |
| `const qosh = (a, b) => a + b;` | `qosh = lambda a, b: a + b` |

# Parametrlar turlari
```python
def darajaga(son, daraja=2):           # standart qiymat
    return son ** daraja
darajaga(5)        # 25
darajaga(2, 10)    # 1024

def anketa(ism, yosh, shahar="Toshkent"):
    print(f"{ism}, {yosh} yosh, {shahar}")
anketa(yosh=15, ism="Laylo", shahar="Xiva")   # nomlangan — tartib muhim emas

def ortacha(*sonlar):                  # istalgan sonli argument → tuple
    return sum(sonlar) / len(sonlar)
ortacha(4, 5, 3, 5)                    # 4.25
```

# lambda va scope
```python
kvadrat = lambda x: x * x
kvadrat(7)    # 49

oquvchilar = [("Ali", 85), ("Vali", 92), ("Gani", 78)]
oquvchilar.sort(key=lambda o: o[1], reverse=True)   # ball bo'yicha

maktab = "5-maktab"          # global
def tanishtir():
    ism = "Laylo"            # lokal
    print(ism, maktab)       # ✅
# print(ism)                 ❌ NameError
```
- **Yaxshi funksiya:** ma'lumotni parametrdan oladi, natijani `return` qiladi; bitta vazifa; nomi fe'l

# datetime — sana va vaqt
```python
from datetime import datetime, date, timedelta

hozir = datetime.now()
hozir.strftime("%d.%m.%Y %H:%M")     # 06.10.2026 14:35
```

# json — ma'lumotlarni saqlash
```python
import json
oquvchilar = [{"ism": "Ali", "baho": 5}, {"ism": "Zarina", "baho": 4}]

with open("oquvchilar.json", "w", encoding="utf-8") as f:
    json.dump(oquvchilar, f, ensure_ascii=False, indent=2)

with open("oquvchilar.json", encoding="utf-8") as f:
    qayta = json.load(f)

json.dumps({"ok": True})     # matnga: '{"ok": true}'
json.loads('{"ok": true}')   # matndan dict'ga
```
- JSON — dasturlar o'rtasida ma'lumot almashishning eng mashhur formati
- `ensure_ascii=False` — o'zbekcha harflar o'qiladigan holda; `indent=2` — chiroyli
