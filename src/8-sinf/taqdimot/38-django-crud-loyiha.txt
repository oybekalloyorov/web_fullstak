# 1. Forma
```python blog/forms.py
class MaqolaForm(forms.ModelForm):
    class Meta:
        model = Maqola
        fields = ["sarlavha", "kategoriya", "matn", "chop_etilgan"]

    def clean_sarlavha(self):
        s = self.cleaned_data["sarlavha"].strip()
        if len(s) < 5:
            raise forms.ValidationError("Sarlavha kamida 5 belgidan iborat bo'lsin")
        return s
```

# 2. Read: ro'yxat (sahifalash) va batafsil
```python
from django.core.paginator import Paginator

def maqola_royxat(request):
    maqolalar = Maqola.objects.filter(chop_etilgan=True).select_related("kategoriya", "muallif")
    sahifa = Paginator(maqolalar, 6).get_page(request.GET.get("page"))   # 6 tadan
    return render(request, "blog/maqola_list.html", {"sahifa": sahifa})

def maqola_batafsil(request, id):
    maqola = get_object_or_404(Maqola, id=id)
    if not maqola.chop_etilgan and maqola.muallif != request.user:
        raise Http404                       # qoralama — faqat muallifga
    return render(request, "blog/maqola_detail.html", {"maqola": maqola})
```
```django

{{ sahifa.number }} / {{ sahifa.paginator.num_pages }}
{% if sahifa.has_next %}<a href="?page={{ sahifa.next_page_number }}">Keyingi →</a>{% endif %}
```

# 3. Create va Update — bitta forma
```python
@login_required
def maqola_yaratish(request):
    form = MaqolaForm(request.POST or None)
    if request.method == "POST" and form.is_valid():
        maqola = form.save(commit=False)
        maqola.muallif = request.user
        maqola.save()
        messages.success(request, "Maqola chop etildi! 🎉")
        return redirect(maqola)
    return render(request, "blog/maqola_form.html", {"form": form, "sarlavha": "Yangi maqola"})

@login_required
def maqola_tahrirlash(request, id):
    maqola = get_object_or_404(Maqola, id=id)
    if maqola.muallif != request.user:
        return HttpResponseForbidden("Bu maqolani tahrirlash huquqingiz yo'q")
    form = MaqolaForm(request.POST or None, instance=maqola)   # ⬅ instance!
    if request.method == "POST" and form.is_valid():
        form.save()
        return redirect(maqola)
    return render(request, "blog/maqola_form.html", {"form": form, "sarlavha": "Tahrirlash"})
```

# 4. Delete — tasdiqlash va POST
```python
@login_required
def maqola_ochirish(request, id):
    maqola = get_object_or_404(Maqola, id=id, muallif=request.user)
    if request.method == "POST":
        maqola.delete()
        messages.success(request, "Maqola o'chirildi")
        return redirect("blog:bosh")
    return render(request, "blog/maqola_confirm_delete.html", {"maqola": maqola})
```
```django maqola_confirm_delete.html
<h1>O'chirishni tasdiqlang</h1>
<p>«{{ maqola.sarlavha }}» maqolasini o'chirmoqchimisiz? Qaytarib bo'lmaydi.</p>
<form method="post">
  {% csrf_token %}
  <button class="tugma qizil">Ha, o'chirish</button>
  <a href="{{ maqola.get_absolute_url }}">Yo'q, ortga</a>
</form>
```

# 5. CBV: yaratish, tahrirlash, o'chirish
```python
class MaqolaCreate(LoginRequiredMixin, CreateView):
    model = Maqola
    form_class = MaqolaForm
    def form_valid(self, form):
        form.instance.muallif = self.request.user
        return super().form_valid(form)

class MuallifMixin(UserPassesTestMixin):
    def test_func(self):
        return self.get_object().muallif == self.request.user

class MaqolaUpdate(LoginRequiredMixin, MuallifMixin, UpdateView):
    model = Maqola
    form_class = MaqolaForm
    pk_url_kwarg = "id"

class MaqolaDelete(LoginRequiredMixin, MuallifMixin, DeleteView):
    model = Maqola
    pk_url_kwarg = "id"
    success_url = reverse_lazy("blog:bosh")
```

# Tayyor CBV'lar
| Klass | Nima qiladi |
|---|---|
| `ListView` | Ro'yxat + sahifalash |
| `DetailView` | Bitta obyekt (yo'q bo'lsa 404) |
| `CreateView` | Bo'sh forma → tekshirish → yaratish |
| `UpdateView` | Eski qiymatlar bilan forma → saqlash |
| `DeleteView` | Tasdiqlash + POST bilan o'chirish |
| `LoginRequiredMixin` | `@login_required` ning CBV varianti |
| `UserPassesTestMixin` | `test_func()` False → 403 |
- ~80 qator view → ~40 qator
- Shablon nomlari avtomatik: `model_list.html`, `model_detail.html`, `model_form.html`, `model_confirm_delete.html`
