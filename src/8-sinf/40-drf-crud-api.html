2-daraja: Generic view'lar

from rest_framework import generics
from .models import Maqola
from .serializers import MaqolaSerializer


class MaqolaListCreate(generics.ListCreateAPIView):          # GET (ro'yxat) + POST (yaratish)

    serializer_class = MaqolaSerializer

    def perform_create(self, serializer):                    # saqlashdan oldin muallifni qo'shamiz
        serializer.save(muallif=self.request.user)

    queryset = Maqola.objects.all()
    serializer_class = MaqolaSerializer

Ruxsatlar (permissions)

from rest_framework import permissions


class MuallifYokiFaqatOqish(permissions.BasePermission):
    """O'qish — hammaga; o'zgartirish va o'chirish — faqat muallifga."""

    def has_object_permission(self, request, view, obj):
        if request.method in permissions.SAFE_METHODS:       # GET, HEAD, OPTIONS — "xavfsiz" metodlar
            return True
        return obj.muallif == request.user
from rest_framework import permissions
from .permissions import MuallifYokiFaqatOqish


class MaqolaDetail(generics.RetrieveUpdateDestroyAPIView):
    queryset = Maqola.objects.all()
    serializer_class = MaqolaSerializer
    permission_classes = [permissions.IsAuthenticatedOrReadOnly, MuallifYokiFaqatOqish]

3-daraja: ViewSet va Router

from rest_framework import viewsets, permissions, filters
from .models import Maqola, Kategoriya
from .serializers import MaqolaSerializer, KategoriyaSerializer
from .permissions import MuallifYokiFaqatOqish


class MaqolaViewSet(viewsets.ModelViewSet):          # list, create, retrieve, update, partial_update, destroy
    serializer_class = MaqolaSerializer
    permission_classes = [permissions.IsAuthenticatedOrReadOnly, MuallifYokiFaqatOqish]
    filter_backends = [filters.SearchFilter, filters.OrderingFilter]
    search_fields = ["sarlavha", "matn"]                    # ?search=futbol
    ordering_fields = ["yaratilgan", "korishlar"]           # ?ordering=-korishlar

    def get_queryset(self):
        qs = Maqola.objects.filter(chop_etilgan=True).select_related("kategoriya", "muallif")
        kategoriya = self.request.query_params.get("kategoriya")   # ?kategoriya=sport
        if kategoriya:
            qs = qs.filter(kategoriya__slug=kategoriya)
        return qs

    def perform_create(self, serializer):
        serializer.save(muallif=self.request.user)


class KategoriyaViewSet(viewsets.ReadOnlyModelViewSet):   # faqat o'qish: list + retrieve
    queryset = Kategoriya.objects.all()
    serializer_class = KategoriyaSerializer
from rest_framework.routers import DefaultRouter
from blog.api_views import MaqolaViewSet, KategoriyaViewSet

router = DefaultRouter()
router.register("maqolalar", MaqolaViewSet, basename="maqola")
router.register("kategoriyalar", KategoriyaViewSet)

urlpatterns = [
    path("admin/", admin.site.urls),
    path("api/", include(router.urls)),         # ⬅ barcha API URL'lari avtomatik
    path("api-auth/", include("rest_framework.urls")),   # brauzerdagi API'da "Log in" tugmasi
    path("", include("blog.urls")),
]

3-qadam uchun maslahat 
class MahsulotViewSet(viewsets.ModelViewSet):
    queryset = Mahsulot.objects.filter(faol=True)
    serializer_class = MahsulotSerializer

    def get_permissions(self):
        if self.action in ("list", "retrieve"):
            return [permissions.AllowAny()]
        return [permissions.IsAdminUser()]

Ko'p uchraydigan xatolar

  • ❌ Brauzer konsolida blocked by CORS policy — corsheaders o'rnatilmagan, middleware ro'yxat boshida emas yoki front-end manzili CORS_ALLOWED_ORIGINS da yo'q (port ham mos bo'lishi kerak!).
  • ❌ ViewSet'da get_queryset ishlatilganda router'da basename unutilgan.
  • ❌ Generic view URL'ida <int:id> — generic'lar standart holatda pk kutadi.
  • ❌ fetch natijasini .json() siz ishlatish yoki await ni unutish.

O'zingizni tekshiring

Bitta klassda list, create, retrieve, update, destroy amallarini beruvchi DRF klassi?

  1. APIView
  2. ListAPIView
  3. ModelViewSet
  4. Serializer

ViewSet uchun URL'larni avtomatik yaratuvchi vosita?

  1. Router (DefaultRouter)
  2. path()
  3. Paginator
  4. Middleware

SAFE_METHODS qaysi metodlarni o'z ichiga oladi?

  1. POST, PUT
  2. DELETE
  3. Hammasi
  4. GET, HEAD, OPTIONS

Netlify'dagi JS sayt Django API'ga so'rov yuborganda "CORS" xatosi chiqdi. Nima qilish kerak?

  1. JavaScript kodini o'chirish
  2. Django'da django-cors-headers ni sozlab, front-end manzilini ruxsat etilganlarga qo'shish
  3. Brauzerni almashtirish
  4. API'ni HTML'ga aylantirish

API'da qidiruv uchun DRF'ning qaysi vositasi ishlatiladi?

  1. Paginator
  2. OrderingFilter
  3. SearchFilter + search_fields
  4. Router
