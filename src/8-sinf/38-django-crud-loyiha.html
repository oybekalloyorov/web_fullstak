3. View'lar (funksiya asosida)

from django.contrib import messages
from django.contrib.auth.decorators import login_required
from django.core.paginator import Paginator
from django.db.models import F
from django.http import HttpResponseForbidden
from django.shortcuts import get_object_or_404, redirect, render

from .forms import MaqolaForm
from .models import Maqola


# ---------- READ: ro'yxat ----------
def maqola_royxat(request):
    maqolalar = (Maqola.objects.filter(chop_etilgan=True)
                 .select_related("kategoriya", "muallif"))
    sahifalovchi = Paginator(maqolalar, 6)                  # har sahifada 6 ta
    sahifa = sahifalovchi.get_page(request.GET.get("page")) # ?page=2
    return render(request, "blog/maqola_list.html", {"sahifa": sahifa})


# ---------- READ: batafsil ----------
def maqola_batafsil(request, id):
    maqola = get_object_or_404(Maqola, id=id)
    # Chop etilmagan maqolani faqat muallifi ko'ra oladi
    if not maqola.chop_etilgan and maqola.muallif != request.user:
        return HttpResponseForbidden("Bu maqola hali chop etilmagan")
    Maqola.objects.filter(id=id).update(korishlar=F("korishlar") + 1)
    return render(request, "blog/maqola_detail.html", {"maqola": maqola})


# ---------- CREATE ----------
@login_required
def maqola_yaratish(request):
    if request.method == "POST":
        form = MaqolaForm(request.POST)
        if form.is_valid():
            maqola = form.save(commit=False)
            maqola.muallif = request.user
            maqola.save()
            messages.success(request, "✅ Maqola yaratildi")
            return redirect(maqola)                          # get_absolute_url() ga
    else:
        form = MaqolaForm()
    return render(request, "blog/maqola_form.html", {"form": form, "sarlavha": "Yangi maqola"})


# ---------- UPDATE ----------
@login_required
def maqola_tahrirlash(request, id):
    maqola = get_object_or_404(Maqola, id=id)
    if maqola.muallif != request.user:                      # 🔒 faqat muallif
        return HttpResponseForbidden("Siz bu maqolani tahrirlay olmaysiz")
    if request.method == "POST":
        form = MaqolaForm(request.POST, instance=maqola)    # instance — mavjud obyektni yangilash
        if form.is_valid():
            form.save()
            messages.success(request, "✅ O'zgarishlar saqlandi")
            return redirect(maqola)
    else:
        form = MaqolaForm(instance=maqola)                  # forma eski qiymatlar bilan to'lgan
    return render(request, "blog/maqola_form.html", {"form": form, "sarlavha": "Tahrirlash"})


# ---------- DELETE ----------
@login_required
def maqola_ochirish(request, id):
    maqola = get_object_or_404(Maqola, id=id)
    if maqola.muallif != request.user:
        return HttpResponseForbidden("Siz bu maqolani o'chira olmaysiz")
    if request.method == "POST":                            # faqat tasdiqlangandan keyin
        maqola.delete()
        messages.success(request, "🗑 Maqola o'chirildi")
        return redirect("blog:bosh")
    return render(request, "blog/maqola_confirm_delete.html", {"maqola": maqola})
Xavfsizlik: ikki qoida
  1. Ruxsatni serverda tekshiring. Shablonda "Tahrirlash" tugmasini yashirish yetarli emas — ayyor foydalanuvchi /maqola/5/tahrirlash/ manzilini qo'lda yozishi mumkin. Shuning uchun view'da if maqola.muallif != request.user.
  2. O'chirish — faqat POST. Agar GET so'rov bilan o'chirsak, oddiy havola (yoki boshqa saytdagi rasm teg!) maqolani o'chirib yuborishi mumkin. Avval tasdiqlash sahifasi, keyin POST forma + CSRF.

MaqolaForm(request.POST, instance=maqola) — yaratish va tahrirlash uchun bitta forma va bitta shablon: instance berilsa — mavjud obyekt yangilanadi, berilmasa — yangisi yaratiladi.

5. Xuddi shu narsa — Class Based View'lar bilan

CRUD shunchalik keng tarqalganki, Django uning uchun tayyor klasslar beradi. Yuqoridagi ~80 qator view kodi ~40 qatorga qisqaradi:

from django.contrib.auth.mixins import LoginRequiredMixin, UserPassesTestMixin
from django.contrib.messages.views import SuccessMessageMixin
from django.urls import reverse_lazy
from django.views.generic import ListView, DetailView, CreateView, UpdateView, DeleteView

from .forms import MaqolaForm
from .models import Maqola


class MaqolaList(ListView):
    queryset = Maqola.objects.filter(chop_etilgan=True).select_related("kategoriya", "muallif")
    paginate_by = 6                         # shablonda: page_obj, object_list
    # shablon (standart): blog/maqola_list.html


class MaqolaDetail(DetailView):
    model = Maqola
    pk_url_kwarg = "id"                     # URL'da  ishlatganimiz uchun
    # shablon: blog/maqola_detail.html, obyekt: {{ object }} yoki {{ maqola }}


class MaqolaCreate(LoginRequiredMixin, SuccessMessageMixin, CreateView):
    model = Maqola
    form_class = MaqolaForm
    success_message = "✅ Maqola yaratildi"
    extra_context = {"sarlavha": "Yangi maqola"}      # shablondagi {{ sarlavha }} uchun

    def form_valid(self, form):
        form.instance.muallif = self.request.user
        return super().form_valid(form)     # saqlash va get_absolute_url() ga o'tish


class FaqatMuallif(UserPassesTestMixin):
    """Faqat muallifga ruxsat beruvchi 'aralashma' (mixin)."""
    def test_func(self):
        return self.get_object().muallif == self.request.user


class MaqolaUpdate(LoginRequiredMixin, FaqatMuallif, SuccessMessageMixin, UpdateView):
    model = Maqola
    form_class = MaqolaForm
    pk_url_kwarg = "id"
    success_message = "✅ O'zgarishlar saqlandi"
    extra_context = {"sarlavha": "Tahrirlash"}


class MaqolaDelete(LoginRequiredMixin, FaqatMuallif, DeleteView):
    model = Maqola
    pk_url_kwarg = "id"
    success_url = reverse_lazy("blog:bosh")
from . import views_cbv as v

urlpatterns = [
    path("", v.MaqolaList.as_view(), name="bosh"),
    path("maqola/yangi/", v.MaqolaCreate.as_view(), name="yaratish"),

Ko'p uchraydigan xatolar

  • ❌ maqola/yangi/ yo'li maqola/<int:id>/ dan keyin yozilsa — muammo yo'q (chunki "yangi" son emas), lekin <str:...> ishlatilsa — "yangi" so'zi id deb olinadi. Aniq yo'llarni doim oldin yozing.
  • ❌ Tahrirlashda instance= unutilgan — yangi nusxa yaratiladi, eskisi o'zgarmaydi.
  • ❌ Ruxsat faqat shablonda tekshirilgan — URL'ni qo'lda yozib, boshqa birovning maqolasini o'chirish mumkin bo'lib qoladi.
  • ❌ GET so'rov bilan o'chirish.
  • ❌ TemplateDoesNotExist: blog/maqola_form.html — CBV standart nomli shablonni qidiryapti; faylni shunday nomlang yoki template_name bering.

O'zingizni tekshiring

ModelForm bilan mavjud obyektni tahrirlash uchun formaga nima beriladi?

  1. id=
  2. object=
  3. instance=
  4. update=True

Nima uchun o'chirish faqat POST so'rov bilan bo'lishi kerak?

  1. POST tezroq ishlaydi
  2. Oddiy havola yoki begona sayt tasodifan/ataylab o'chirib yubormasligi uchun (CSRF himoyasi bilan)
  3. GET so'rovda id bo'lmaydi
  4. Django GET'ni qo'llab-quvvatlamaydi

"Faqat muallif tahrirlay oladi" tekshiruvi qayerda bo'lishi SHART?

  1. View'da (serverda)
  2. Faqat shablonda
  3. CSS'da
  4. JavaScript'da

Ro'yxatni sahifalarga bo'luvchi Django klassi?

  1. Splitter
  2. PageView
  3. Limiter
  4. Paginator

Yangi obyekt yaratuvchi tayyor CBV?

  1. ListView
  2. CreateView
  3. NewView
  4. FormView

CBV'da tizimga kirishni talab qiluvchi aralashma (mixin)?

  1. AuthMixin
  2. UserPassesTestMixin
  3. LoginRequiredMixin
  4. SessionMixin

🎉 IV bob yakunlandi!

  • CRUD = ro'yxat, batafsil, yaratish, tahrirlash, o'chirish — 5 ta URL/view/shablon.
  • ModelForm(request.POST, instance=obj) — yaratish va tahrirlash uchun bitta forma.
  • Ruxsatlar serverda; o'chirish — tasdiqlash + POST.
  • Paginator; tayyor CBV'lar: ListView, DetailView, CreateView, UpdateView, DeleteView + mixinlar.
  • Siz endi to'liq ishlaydigan, xavfsiz veb-ilova yarata olasiz! Keyingi bobda unga API qo'shamiz va internetga chiqaramiz.
