⚠️ SQL injection — eng xavfli xato

Python
# ❌ HECH QACHON BUNDAY QILMANG
ism = input("Ism: ")
cur.execute(f"SELECT * FROM oquvchilar WHERE ism = '{ism}'")
  • Buzg'unchi yozadi: ' OR '1'='1
SQL
SELECT * FROM oquvchilar WHERE ism = '' OR '1'='1'   -- doim rost → HAMMA ma'lumot!
  • Yoki: '; DROP TABLE oquvchilar; -- → jadval o'chadi!
💡
Buyruq qog'ozi: "Ombordan ____ ni olib kel". Ayyor mijoz yozadi: "olma. Keyin seyfdagi pullarni ham". Parametrli so'rov bo'sh joyni faqat mahsulot nomi deb tushunadi.

✅ Parametrli so'rov

Python
ism = input("Ism: ")
cur.execute("SELECT * FROM oquvchilar WHERE ism = %s", (ism,))   # (ism,) — tuple!

cur.execute(
    "INSERT INTO oquvchilar (ism, familiya, sinf_id) VALUES (%s, %s, %s)",
    ("Jasur", "Rahimov", 2),
)
SQL'ga kiradigan har qanday qiymat — faqat %s orqali
  • %s — barcha turlar uchun (son uchun ham), qo'shtirnoqsiz
  • f-string, +, .format() bilan SQL yasamang!

CRUD'ni Python'dan bajarish

Python
with psycopg.connect(DSN) as conn:
    with conn.cursor() as cur:
        cur.execute(
            "INSERT INTO oquvchilar (ism, familiya, sinf_id) VALUES (%s, %s, %s) RETURNING id",
            ("Laylo", "Usmonova", 1))
        yangi_id = cur.fetchone()[0]                                # CREATE

        cur.execute("SELECT ism, familiya FROM oquvchilar WHERE id = %s", (yangi_id,))
        print(cur.fetchone())                                       # READ

        cur.execute("UPDATE oquvchilar SET sinf_id = %s WHERE id = %s", (2, yangi_id))
        print("O'zgardi:", cur.rowcount)                            # UPDATE

        cur.execute("DELETE FROM oquvchilar WHERE id = %s", (yangi_id,))   # DELETE

import os, psycopg
from psycopg.rows import dict_row
from dotenv import load_dotenv
load_dotenv()

def ulanish():
    return psycopg.connect(
        dbname=os.getenv("DB_NAME"),
        user=os.getenv("DB_USER"),
        password=os.getenv("DB_PASSWORD"),
        host=os.getenv("DB_HOST"),
        row_factory=dict_row,
    )

Amaliyot: CRUD dasturi (20-loyiha)

Python
from db import ulanish

def royxat():
    with ulanish() as conn:
        rows = conn.execute("""
            SELECT o.id, o.ism, o.familiya, s.nomi AS sinf,
                   ROUND(AVG(b.baho), 2) AS ortacha
            FROM oquvchilar o
            LEFT JOIN sinflar s ON s.id = o.sinf_id
            LEFT JOIN baholar b ON b.oquvchi_id = o.id
            GROUP BY o.id, s.nomi ORDER BY o.familiya
        """).fetchall()
    for r in rows:
        print(f"{r['id']:<4}{r['ism']:<12}{r['familiya']:<14}{r['sinf'] or '-':<6}{r['ortacha'] or '-'}")

def qidirish(soz):
    with ulanish() as conn:
        return conn.execute(
            "SELECT * FROM oquvchilar WHERE ism ILIKE %s OR familiya ILIKE %s",
            (f"%{soz}%", f"%{soz}%")).fetchall()

Ko'p uchraydigan xatolar

  • ❌ No module named 'psycopg' — venv faol emas yoki paket o'rnatilmagan

  • ❌ (ism) — bu tuple emas! To'g'risi (ism,)
  • ❌ '%s' — qo'shtirnoq ichida; to'g'risi %s
  • ❌ commit qilinmagan — ma'lumot "yo'qoldi"
