  • Izoh: -- bir qator, /* ko'p qator */

Ma'lumot turlari

TurNima uchunMisol
INTEGER / SMALLINTButun son / kichik son14, baho 5
SERIALAvtomatik 1, 2, 3... — id uchun—
NUMERIC(10, 2)Aniq o'nlik — pul12500.50

BOOLEANRost / yolg'onTRUE
DATE / TIMESTAMPSana / sana va vaqt'2012-03-15'

Turni to'g'ri tanlash

  • Pul uchun REAL emas, NUMERIC — kasr sonlar aniq saqlanmaydi: 0.1 + 0.2 ≠ 0.3!
  • Telefon uchun INTEGER emas, VARCHAR(20) — + belgisi va boshidagi 0
  • Sanani matn qilib saqlamang — DATE bilan solishtirish va hisoblash mumkin

CREATE TABLE

SQL
CREATE TABLE sinflar (
    id            SERIAL PRIMARY KEY,
    nomi          VARCHAR(10) NOT NULL UNIQUE,
    sinf_rahbari  VARCHAR(100),
    xona          SMALLINT
);

CREATE TABLE oquvchilar (
    id             SERIAL PRIMARY KEY,
    ism            VARCHAR(50) NOT NULL,
    familiya       VARCHAR(50) NOT NULL,
    tugilgan_sana  DATE,
    telefon        VARCHAR(20),
    email          VARCHAR(100) UNIQUE,
    alochi         BOOLEAN DEFAULT FALSE,
    sinf_id        INTEGER REFERENCES sinflar(id),   -- tashqi kalit!
    yaratilgan     TIMESTAMP DEFAULT NOW()
);

baholar jadvali va CASCADE

SQL
CREATE TABLE baholar (
    id          SERIAL PRIMARY KEY,
    oquvchi_id  INTEGER NOT NULL REFERENCES oquvchilar(id) ON DELETE CASCADE,
    fan         VARCHAR(50) NOT NULL,
    baho        SMALLINT NOT NULL CHECK (baho BETWEEN 2 AND 5),
    sana        DATE NOT NULL DEFAULT CURRENT_DATE
);
  • ON DELETE CASCADE — o'quvchi o'chirilsa, baholari ham o'chadi (yetim yozuv qolmaydi)
⚠️
Tartib: avval sinflar, keyin oquvchilar, keyin baholar — aks holda "relation does not exist"

ALTER va DROP

SQL
ALTER TABLE oquvchilar ADD COLUMN manzil TEXT;
ALTER TABLE oquvchilar RENAME COLUMN manzil TO yashash_joyi;
ALTER TABLE oquvchilar ALTER COLUMN telefon TYPE VARCHAR(30);
ALTER TABLE oquvchilar DROP COLUMN yashash_joyi;

DROP TABLE baholar;          -- ⚠️ barcha ma'lumotlari bilan!
⚠️
DROP TABLE / DROP DATABASE — "Ctrl+Z" yo'q! Ikki marta o'ylang va zaxira nusxa oling (29-mavzu)
