break va continue

JavaScript
for (let i = 1; i <= 10; i++) {
  if (i === 5) break;      // siklni darhol TO'XTATISH
  console.log(i);          // 1 2 3 4
}

for (let i = 1; i <= 10; i++) {
  if (i % 3 === 0) continue;   // shu qadamni O'TKAZIB yuborish
  console.log(i);              // 1 2 4 5 7 8 10
}

Funksiya nima?

Parametrlar

JavaScript
function salomlash(ism, sinf) {       // parametrlar
  console.log(`Salom, ${ism}! Siz ${sinf}-sinfdasiz.`);
}

salomlash("Bobur", 9);

xabar("Yangi dars", "📢");             // 📢 Yangi dars

return — natija qaytarish

JavaScript
function kvadrat(son) {
  return son * son;
}
const a = kvadrat(5);                  // 25
const b = kvadrat(3) + kvadrat(4);     // 25

function bahoAniqla(ball) {
  if (ball >= 86) return 5;   // return funksiyani TUGATADI
  if (ball >= 71) return 4;
  if (ball >= 56) return 3;
  return 2;
}
  • console.log — faqat ekranga; return — qiymat qaytaradi, uni ishlatish mumkin
  • return siz funksiya undefined qaytaradi

Arrow funksiyalar

JavaScript
function qoshish(a, b) {          // oddiy
  return a + b;
}

const qoshish2 = (a, b) => {      // arrow
  return a + b;
};

const qoshish3 = (a, b) => a + b; // bitta ifoda — return avtomatik
const ikkilantir = son => son * 2; // bitta parametr — qavssiz

Amaliyot: kalkulyator (9-loyiha)

JavaScript
function hisobla(a, amal, b) {
  switch (amal) {
    case "+": return a + b;
    case "-": return a - b;
    case "*": return a * b;
    case "/":
      if (b === 0) return "Nolga bo'lib bo'lmaydi!";
      return a / b;
    default: return "Noma'lum amal";
  }
}
let davom = true;
while (davom) {
  const a = Number(prompt("1-son:"));
  const amal = prompt("Amal (+ - * /):");
  const b = Number(prompt("2-son:"));
  alert(`Natija: ${hisobla(a, amal, b)}`);
  davom = confirm("Yana hisoblaymizmi?");
}

Amaliyot: "Sonni top" (10-loyiha)

JavaScript
const yashirin = Math.floor(Math.random() * 100) + 1;
let urinish = 0, topildi = false;

while (urinish < 7 && !topildi) {
  const taxmin = Number(prompt("1..100 oralig'ida son:"));
  urinish++;
  if (taxmin === yashirin) topildi = true;
  else if (taxmin < yashirin) alert("⬆️ Kattaroq!");
  else alert("⬇️ Kichikroq!");
}
alert(topildi ? `🎉 ${urinish} urinishda topdingiz!` : `😢 Son: ${yashirin}`);
