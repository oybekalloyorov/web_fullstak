Admin'ni sozlash

Python
@admin.register(Kategoriya)
class KategoriyaAdmin(admin.ModelAdmin):
    list_display = ["nomi", "slug"]
    prepopulated_fields = {"slug": ("nomi",)}

@admin.register(Maqola)
class MaqolaAdmin(admin.ModelAdmin):
    list_display = ["sarlavha", "kategoriya", "muallif", "chop_etilgan", "korishlar"]
    list_filter = ["chop_etilgan", "kategoriya", "yaratilgan"]
    search_fields = ["sarlavha", "matn"]
    list_editable = ["chop_etilgan"]
    readonly_fields = ["korishlar"]

Oddiy Form: "Aloqa"

blog/forms.py
from django import forms

class AloqaForm(forms.Form):
    ism = forms.CharField(label="Ismingiz", max_length=100)
    email = forms.EmailField(label="Email")
    mavzu = forms.ChoiceField(choices=[("taklif", "Taklif"), ("xato", "Xato haqida")])
    xabar = forms.CharField(widget=forms.Textarea(attrs={"rows": 5}), min_length=10)

    def clean_ism(self):                       # o'z tekshiruvimiz
        ism = self.cleaned_data["ism"].strip()
        if any(b.isdigit() for b in ism):
            raise forms.ValidationError("Ismda raqam bo'lmasligi kerak")
        return ism

View: forma sxemasi

Python
from django.contrib import messages

def aloqa(request):
    if request.method == "POST":
        form = AloqaForm(request.POST)
        if form.is_valid():                         # ✅ tekshiruvdan o'tdi
            data = form.cleaned_data
            messages.success(request, "Rahmat! Xabaringiz yuborildi.")
            return redirect("blog:aloqa")           # qayta yuborilmasin
    else:
        form = AloqaForm()                          # GET — bo'sh forma
    return render(request, "blog/aloqa.html", {"form": form})

CSRF — yovuz sayt sizning brauzeringiz nomidan boshqa saytga yashirin forma yuboradi ("barcha maqolalarni o'chir"). {% csrf_token %} yashirin maxfiy kod qo'shadi — Django faqat shu kod to'g'ri bo'lsa qabul qiladi.

ModelForm — modeldan avtomatik forma

Python
class MaqolaForm(forms.ModelForm):
    class Meta:
        model = Maqola
        fields = ["sarlavha", "kategoriya", "matn"]
        widgets = {"matn": forms.Textarea(attrs={"rows": 8})}

@login_required
def maqola_yozish(request):
    if request.method == "POST":
        form = MaqolaForm(request.POST)
        if form.is_valid():
            maqola = form.save(commit=False)     # hali bazaga yozilmadi
            maqola.muallif = request.user        # formada yo'q maydon
            maqola.save()
            return redirect(maqola)              # get_absolute_url
    else:
        form = MaqolaForm()
    return render(request, "blog/yozish.html", {"form": form})

Ro'yxatdan o'tish

Python
from django.contrib.auth import login
from django.contrib.auth.forms import UserCreationForm

def royxatdan_otish(request):
    if request.method == "POST":
        form = UserCreationForm(request.POST)
        if form.is_valid():
            user = form.save()              # parol avtomatik xeshlanadi
            login(request, user)
            return redirect("blog:bosh")
    else:
        form = UserCreationForm()
    return render(request, "blog/royxat.html", {"form": form})
